extends SceneTree

func _initialize() -> void:
	call_deferred("run")

func frames(n: int) -> void:
	for i in n: await process_frame

func clear_guidance(session, game) -> void:
	var pending: Dictionary = session.pending_guidance()
	var tries := 0
	while not pending.is_empty() and tries < 20:
		assert(session.dismiss_guidance(str(pending.kind), str(pending.id)).ok)
		pending = session.pending_guidance()
		tries += 1
	game._sync_guidance()

func shot(name: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("/tmp/space-idle-runtime472/style-followup/" + name + "-1440x900.png")

func find_button(node: Node, caption: String) -> Button:
	if node is Button and node.text == caption: return node
	for child in node.get_children():
		var found := find_button(child, caption)
		if found != null: return found
	return null

func find_path_field(node: Node) -> LineEdit:
	if node is LineEdit and node.placeholder_text.begins_with("存档目录"): return node
	for child in node.get_children():
		var found := find_path_field(child)
		if found != null: return found
	return null

func hold(button: Button) -> void:
	var point := button.get_global_rect().get_center()
	Input.warp_mouse(point)
	await frames(3)
	var event := InputEventMouseButton.new()
	event.button_index = MOUSE_BUTTON_LEFT
	event.pressed = true
	event.position = point
	event.global_position = point
	root.push_input(event)
	await frames(2)

func release(button: Button) -> void:
	var point := button.get_global_rect().get_center()
	var event := InputEventMouseButton.new()
	event.button_index = MOUSE_BUTTON_LEFT
	event.position = point
	event.global_position = point
	root.push_input(event)
	await frames(2)

func run() -> void:
	root.size = Vector2i(1440, 900)
	root.content_scale_size = Vector2i(1440, 900)
	var game: Node2D = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await frames(45)
	assert(game.session != null and game.startup_error.is_empty())
	clear_guidance(game.session, game)
	game._refresh_feature_visibility()
	game._zone_shell._route_buttons["settings"].pressed.emit()
	await frames(6)
	clear_guidance(game.session, game)
	assert(game._music_toggle != null and game._effects_toggle != null)
	await frames(3)
	await shot("public-source-settings-normal")
	var motion := InputEventMouseMotion.new()
	motion.position = game._music_toggle.get_global_rect().get_center()
	motion.global_position = motion.position
	root.push_input(motion)
	await frames(3)
	print("MUSIC_HOVERED ", game._music_toggle.is_hovered(), " DRAW ", game._music_toggle.get_draw_mode(), " STYLE ", game._music_toggle.get_theme_stylebox("hover").get_class())
	await shot("public-source-settings-hover")
	game._music_toggle.grab_focus()
	await frames(2)
	await shot("public-source-settings-focus")
	var select: OptionButton = game._damage_option
	var select_motion := InputEventMouseMotion.new()
	select_motion.position = select.get_global_rect().get_center()
	select_motion.global_position = select_motion.position
	root.push_input(select_motion)
	await frames(3)
	print("SELECT_HOVERED ", select.is_hovered(), " DRAW ", select.get_draw_mode(), " STYLE ", select.get_theme_stylebox("hover").get_class())
	await shot("public-source-select-hover")
	select.grab_focus()
	await frames(2)
	await shot("public-source-select-focus")
	select.release_focus()
	var path_field: LineEdit = find_path_field(game)
	assert(path_field != null)
	path_field.grab_focus()
	await frames(2)
	await shot("public-source-input-focus")
	path_field.release_focus()
	game._music_toggle.release_focus()
	game._effects_toggle.disabled = true
	await frames(2)
	await shot("public-source-settings-disabled")
	game._effects_toggle.disabled = false
	game._zone_shell._show_balances()
	await frames(4)
	await shot("public-source-resources")
	game._zone_shell._details.hide()
	var resources := find_button(game._zone_shell, "资源")
	var save := find_button(game, "保存进度")
	assert(resources != null and save != null)
	await hold(resources)
	await shot("public-source-top-held")
	await release(resources)
	game._zone_shell._details.hide()
	await hold(save)
	await shot("public-source-action-held")
	save.disabled = true
	await release(save)
	print("PUBLIC_SOURCE settings_and_resources captured; music_checked=", game._music_toggle.button_pressed)
	if game.session.save_files != null: game.session.save_files.close(2000)
	game.queue_free()
	await frames(3)
	quit()
