extends SceneTree
const OUT := "res://docs/ui-redesign-plan/shared-host-shell/runtime-clear-zone/evidence/"
func _initialize() -> void:
	call_deferred("run")
func run() -> void:
	root.size = Vector2i(1440, 900)
	var game = load("res://src/main.gd").new()
	root.add_child(game)
	for i in 90: await process_frame
	if not game.startup_error.is_empty():
		print("DROP_SEQ startup_failed ", game.startup_error)
		quit(1)
		return
	game._skip_tutorial = true
	game._sync_guidance()
	for frame in 1200:
		await process_frame
		if not game.session.economy.drop_markers().is_empty():
			for offset in [0, 10]:
				if offset > 0:
					for j in (offset - (0 if offset == 10 else 10 if offset == 30 else 30 if offset == 60 else 60)):
						await process_frame
				var marks: Array = game.session.economy.drop_markers()
				if marks.is_empty():
					print("DROP_SEQ gone ",offset)
					break
				await RenderingServer.frame_post_draw
				root.get_texture().get_image().save_png(OUT + "resource-material-drop-%03d.png" % offset)
				print("DROP_SEQ frame=",frame," offset=",offset," resource=",marks[0].resource," x=",marks[0].x," y=",marks[0].y)
			quit()
			return
	print("DROP_SEQ none")
	quit()
