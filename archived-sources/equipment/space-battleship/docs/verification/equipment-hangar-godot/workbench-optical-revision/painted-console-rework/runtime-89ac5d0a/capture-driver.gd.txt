func tree_ids(node:Node)->Array:

func material_ids()->Array:
    var ids:Array=[]
    var buttons:Array=[panel._purchase,panel._quantity_button,panel._install,panel._batch_submit,panel._batch_quantity,panel._batch_close]
    for key in panel._choices:buttons.append(panel._choices[key].button)
    for button in buttons:
        for variant in button.get_meta("material_variants"):
            for state in variant:ids.append(variant[state].get_instance_id())
    return ids
