extends SceneTree
var api
var galaxy_script
var run_script
var config
var failures: Array = []
func check(condition: bool, label: String) -> void:
	if not condition: failures.append(label)
func amount(text: String) -> Dictionary:
	return api.apply("from_decimal", [text]).value
func equal_amount(value: Dictionary, expected: String) -> bool:
	return api.apply("compare", [value, amount(expected)]).scalar == 0
func make_galaxy():
	return galaxy_script.new(config.galaxy.galaxy, config.galaxy_build.galaxy_build, config.galaxy_config.galaxy_config, api, run_script.new())
func _initialize() -> void:
	var loaded = GDExtensionManager.load_extension("D:/CodexBuild/space_idle/native-stable/space_native.gdextension")
	if loaded not in [GDExtensionManager.LOAD_STATUS_OK, GDExtensionManager.LOAD_STATUS_ALREADY_LOADED]:
		printerr("extension failed: ", loaded)
		quit(1)
		return
	call_deferred("run")
func run() -> void:
	api = ClassDB.instantiate("FoundationAmount")
	galaxy_script = load("C:/Users/Administrator/AppData/Local/Temp/galaxy-ledger-rebuild/baseline-galaxy.gd")
	run_script = load("res://src/systems/run_state.gd")
	config = JSON.parse_string(FileAccess.get_file_as_string("C:/Users/Administrator/AppData/Local/Temp/galaxy-ledger-rebuild/config.json"))
	var owner = make_galaxy()
	var fresh: Dictionary = owner.contributions()
	check(equal_amount(fresh.values.crew_exp, "1") and not fresh.growth_changed, "fresh default")
	var saved: Dictionary = owner.save_state()
	var region: Dictionary = saved.regions.galaxy_1
	var entries: Array = [["colony_ring",1],["colony_ring",2],["orbital_shipyard",3],["stellar_energy_array",4],["crystal_refinery",5],["interstellar_refinery",2],["heavy_element_refinery",3]]
	for i in entries.size():
		var node: Dictionary = region.nodes[str(i)]
		node.building = entries[i][0]
		node.level = entries[i][1]
		node.state = "active"
		node.explored = true
		node.explore_remaining = 0.0
		region.total_levels += node.level
	region.active_count = entries.size()
	region.explored_count = entries.size() + 1
	region.nodes["7"].building = "colony_ring"
	region.nodes["7"].state = "constructing"
	region.nodes["7"].explored = true
	region.nodes["7"].explore_remaining = 0.0
	region.nodes["7"].construction_remaining = 30.0
	region.active_node = "7"
	check(owner.restore_state(saved).ok, "restore admitted")
	var untouched: Dictionary = owner.save_state()
	var restored: Dictionary = owner.contributions()
	for pair in [["crew_exp","1.3"],["equipment_value","1.3"],["charge_max","1.4"],["gem_fragment","1.5"]]:
		check(equal_amount(restored.values[pair[0]], pair[1]), "restored " + pair[0])
	check(restored.growth_changed, "restored growth invalidation")
	check(owner.save_state() == untouched, "rebuild preserves state")
	var payment: Dictionary = owner.income("galaxy_1", {"1": amount("10"), "2": amount("20")}, 10.0)
	check(payment.ok and equal_amount(payment.amounts["1"], "10") and equal_amount(payment.amounts["2"], "30"), "regional iron uranium income")
	var again: Dictionary = owner.contributions()
	check(again.values == restored.values and not again.growth_changed, "repeat no double count")
	var buildings: Dictionary = {}
	for row in config.galaxy_build.galaxy_build: buildings[row.key] = row
	var partial: Array = [{"region":"galaxy_1", "node":"0"}]
	var ledger: Dictionary = galaxy_script._contributions(saved.regions, buildings, api, {}, partial)
	check(equal_amount(ledger.values.crew_exp, "1.3") and ledger.values == restored.values, "first partial changes rebuilds all")
	check(partial.size() == 1, "caller changes preserved")
	var prior_values: Dictionary = ledger.values.duplicate(true)
	var prior_sources: Dictionary = ledger._sources.duplicate(true)
	saved.regions.galaxy_1.nodes["1"].level = 3
	var advanced: Dictionary = galaxy_script._contributions(saved.regions, buildings, api, ledger, [{"region":"galaxy_1", "node":"1"}])
	check(equal_amount(advanced.values.crew_exp, "1.4") and advanced.growth_changed, "incremental replaces source")
	check(ledger.values == prior_values and ledger._sources == prior_sources, "published prior ledger preserved")
	check(owner.restore_state(untouched).ok and equal_amount(owner.contributions().values.crew_exp,"1.3"), "second restore resets sources")
	if failures.is_empty():
		print("PASS: fresh, restore all growth types, non-active exclusion, income, no state mutation, repeat idempotence, first partial changes, incremental replacement, prior snapshot, second restore")
	else: printerr("FAIL: ", failures)
	quit(0 if failures.is_empty() else 1)
