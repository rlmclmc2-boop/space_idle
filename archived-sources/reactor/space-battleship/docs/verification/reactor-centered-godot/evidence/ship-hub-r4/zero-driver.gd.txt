extends SceneTree
func _initialize() -> void:
	call_deferred("run")
func frames(count: int) -> void:
	for i in count: await process_frame
	await RenderingServer.frame_post_draw
func run() -> void:
	root.size = Vector2i(1440, 900)
	root.content_scale_size = Vector2i(1440, 900)
	var game: Node2D = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await frames(28)
	var session = game.session
	var saved: Dictionary = session.campaign_system.save_state()
	saved.grants["feature:reactor"] = true
	saved.grants.erase("reactor_module:condensation")
	assert(session.campaign_system.restore_state(saved).ok)
	session.reactor_system.refresh_availability(session.campaign_system.available)
	assert(session.reactor_system.refresh(session.reactor_inputs(), session.publish_reactor).ok)
	var pending: Dictionary = session.pending_guidance()
	var attempts := 0
	while not pending.is_empty() and attempts < 20:
		assert(session.dismiss_guidance(str(pending.kind), str(pending.id)).ok)
		pending = session.pending_guidance()
		attempts += 1
	game._sync_guidance()
	game._select_left_module("reactor")
	await frames(8)
	var zero := {"weapons":"0", "defence":"0", "smelting":"0"}
	var result: Dictionary = session.reactor_system.apply_allocation_plan(zero, session.reactor_inputs(), session.publish_reactor)
	print("ZERO_APPLY ", result)
	assert(result.get("ok", false))
	await frames(8)
	var model: Dictionary = session.reactor_system.presentation_model()
	var scene = game.reactor_scene
	print("ZERO_MODEL load=", model.load, " rows=", model.modules.map(func(row): return [row.id, row.load]),
		" core_visible=", scene._stage._fields["core"].visible)
	assert(model.load == 0.0 and model.modules.size() == 3)
	assert(not scene._stage._fields["core"].visible)
	for id in ["weapons", "defence", "smelting"]:
		assert(not scene._stage._fields[id].visible and not scene._stage._connections[id].visible)
	root.get_texture().get_image().save_png("/tmp/reactor-hub-three-real-zero.png")
	quit()
