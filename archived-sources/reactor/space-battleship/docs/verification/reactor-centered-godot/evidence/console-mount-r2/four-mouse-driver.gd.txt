extends SceneTree
func _initialize() -> void:
	call_deferred("run")
func frames(count: int) -> void:
	for i in count: await process_frame
	await RenderingServer.frame_post_draw
func mouse(point: Vector2, pressed: bool) -> void:
	var event := InputEventMouseButton.new()
	event.button_index = MOUSE_BUTTON_LEFT
	event.pressed = pressed
	event.position = point
	event.global_position = point
	root.push_input(event, false)
func move(point: Vector2) -> void:
	var event := InputEventMouseMotion.new()
	event.position = point
	event.global_position = point
	root.push_input(event, false)
func click(point: Vector2) -> void:
	move(point)
	mouse(point, true)
	mouse(point, false)
func dismiss(game: Node2D) -> void:
	var pending: Dictionary = game.session.pending_guidance()
	var attempts := 0
	while not pending.is_empty() and attempts < 20:
		assert(game.session.dismiss_guidance(str(pending.kind), str(pending.id)).ok)
		pending = game.session.pending_guidance()
		attempts += 1
	game._sync_guidance()
func run() -> void:
	root.size = Vector2i(1440, 900)
	root.content_scale_size = Vector2i(1440, 900)
	var game: Node2D = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await frames(28)
	var session = game.session
	var saved: Dictionary = session.campaign_system.save_state()
	saved.grants["feature:reactor"] = true
	saved.grants["reactor_module:condensation"] = true
	assert(session.campaign_system.restore_state(saved).ok)
	session.reactor_system.refresh_availability(session.campaign_system.available)
	assert(session.reactor_system.refresh(session.reactor_inputs(), session.publish_reactor).ok)
	dismiss(game)
	game._select_left_module("reactor")
	await frames(12)
	var scene = game.reactor_scene
	click(Vector2(369, 833))
	await frames(2)
	assert(scene._preview.get("ok", false))
	click(Vector2(453, 833))
	await frames(6)
	assert(scene.last_result.get("ok", false))
	for id in ["weapons", "defence", "smelting", "condensation"]:
		var before: int = scene._shares[id]
		var y := 336.0 if id in ["weapons", "defence"] else 670.0
		var start := Vector2(96.0 if id in ["weapons", "smelting"] else 676.0, y)
		var finish := Vector2(54.0 if id in ["weapons", "smelting"] else 631.0, y)
		move(start)
		mouse(start, true)
		await frames(2)
		move(finish)
		await frames(2)
		mouse(finish, false)
		await frames(2)
		print("MOUSE_FOUR ", id, " ", before, " -> ", scene._shares[id], " preview=", scene._preview.get("ok"))
		assert(scene._shares[id] != before and scene._preview.get("ok", false))
	dismiss(game)
	move(Vector2(800, 440))
	await frames(5)
	root.get_texture().get_image().save_png("/tmp/reactor-mount-r2-four-mouse-preview.png")
	click(Vector2(453, 833))
	await frames(8)
	print("MOUSE_FOUR_APPLY ", scene.last_result.get("ok"), " modules=", session.reactor_system.presentation_model().modules.size())
	assert(scene.last_result.get("ok", false))
	dismiss(game)
	move(Vector2(800, 440))
	await frames(5)
	root.get_texture().get_image().save_png("/tmp/reactor-mount-r2-four-mouse-applied.png")
	quit()
