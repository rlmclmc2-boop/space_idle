extends SceneTree
func _initialize() -> void:
    call_deferred("run")
func frames(n: int) -> void:
    for i in n: await process_frame
    await RenderingServer.frame_post_draw
func mouse(pos: Vector2, pressed: bool) -> void:
    var event := InputEventMouseButton.new()
    event.button_index = MOUSE_BUTTON_LEFT
    event.pressed = pressed
    event.position = pos
    event.global_position = pos
    root.push_input(event, false)
func move(pos: Vector2) -> void:
    var event := InputEventMouseMotion.new()
    event.position = pos
    event.global_position = pos
    root.push_input(event, false)
func run() -> void:
    root.size = Vector2i(1440,900)
    root.content_scale_size = Vector2i(1440,900)
    var game: Node2D = load("res://src/main.tscn").instantiate()
    root.add_child(game)
    await frames(28)
    var session = game.session
    var saved: Dictionary = session.campaign_system.save_state()
    saved.grants["feature:reactor"] = true
    assert(session.campaign_system.restore_state(saved).ok)
    session.reactor_system.refresh_availability(session.campaign_system.available)
    assert(session.reactor_system.refresh(session.reactor_inputs(), session.publish_reactor).ok)
    var pending: Dictionary = session.pending_guidance()
    var tries := 0
    while not pending.is_empty() and tries < 20:
        assert(session.dismiss_guidance(str(pending.kind), str(pending.id)).ok)
        pending = session.pending_guidance()
        tries += 1
    game._sync_guidance()
    game._select_left_module("reactor")
    await frames(20)
    var scene = game.reactor_scene
    var before: int = scene._shares["weapons"]
    move(Vector2(94,380))
    mouse(Vector2(94,380),true)
    await frames(2)
    move(Vector2(53,380))
    await frames(2)
    mouse(Vector2(53,380),false)
    await frames(3)
    print("MOUSE_SLIDE before=",before," after=",scene._shares["weapons"]," kind=",scene._kind," preview=",scene._preview.get("ok"))
    assert(scene._shares["weapons"] != before and scene._kind == "shares")
    root.get_texture().get_image().save_png("/tmp/space-idle-native472/reactor-mouse-preview-r2.png")
    move(Vector2(470,857))
    mouse(Vector2(470,857),true)
    mouse(Vector2(470,857),false)
    await frames(8)
    print("MOUSE_APPLY result=",scene.last_result," kind=",scene._kind)
    assert(scene.last_result.get("ok",false))
    root.get_texture().get_image().save_png("/tmp/space-idle-native472/reactor-mouse-applied-r2.png")
    var initial_level: int = session.reactor_system.presentation_model().level
    var amount: Dictionary = session.amount_api.apply("from_decimal",["1000"])
    assert(amount.ok)
    assert(session.economy.credit_yield("2",amount.value,session.wall_time,session.game_time).ok)
    await frames(5)
    assert(not scene._purchase.disabled)
    move(Vector2(262,856))
    mouse(Vector2(262,856),true)
    mouse(Vector2(262,856),false)
    await frames(6)
    print("MOUSE_PURCHASE level=",initial_level,"->",session.reactor_system.presentation_model().level," result=",scene.last_result.get("ok"))
    assert(scene.last_result.get("ok",false) and session.reactor_system.presentation_model().level == initial_level + 1)
    root.get_texture().get_image().save_png("/tmp/space-idle-native472/reactor-mouse-purchased-r2.png")
    quit()
