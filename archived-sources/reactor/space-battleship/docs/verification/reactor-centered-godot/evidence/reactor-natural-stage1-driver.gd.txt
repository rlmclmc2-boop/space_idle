extends SceneTree
func _initialize() -> void:
    call_deferred("run")
func run() -> void:
    root.size = Vector2i(1440,900)
    root.content_scale_size = Vector2i(1440,900)
    var game: Node2D = load("res://src/main.tscn").instantiate()
    root.add_child(game)
    for i in 30: await process_frame
    var session = game.session
    var pending: Dictionary = session.pending_guidance()
    var tries := 0
    while not pending.is_empty() and tries < 20:
        assert(session.dismiss_guidance(str(pending.kind),str(pending.id)).ok)
        pending = session.pending_guidance()
        tries += 1
    game._sync_guidance()
    print("NATURAL_INITIAL available=",session.campaign_system.available("feature","reactor")," progress=",session.campaign_system.progress_facts()," scenery=",session.campaign_system.scenery_facts()," running=",session.run_state.is_running())
    var steps := 0
    while steps < 30000 and session.run_state.is_running() and not session.campaign_system.available("feature","reactor"):
        session._step(1.0/60.0)
        steps += 1
        if steps % 1200 == 0:
            print("NATURAL_PROGRESS steps=",steps," progress=",session.campaign_system.progress_facts()," scenery=",session.campaign_system.scenery_facts())
            await process_frame
    print("NATURAL_FINAL steps=",steps," available=",session.campaign_system.available("feature","reactor")," progress=",session.campaign_system.progress_facts()," scenery=",session.campaign_system.scenery_facts()," running=",session.run_state.is_running()," reason=",session.run_state.reason())
    if session.campaign_system.available("feature","reactor"):
        pending = session.pending_guidance()
        tries = 0
        while not pending.is_empty() and tries < 20:
            assert(session.dismiss_guidance(str(pending.kind),str(pending.id)).ok)
            pending = session.pending_guidance()
            tries += 1
        game._sync_guidance()
        game._select_left_module("reactor")
        for i in 6: await process_frame
        await RenderingServer.frame_post_draw
        root.get_texture().get_image().save_png("/tmp/space-idle-native472/reactor-natural-stage1-r4.png")
    quit()
