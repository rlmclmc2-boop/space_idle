extends SceneTree

func _initialize() -> void:
	call_deferred("run")

func _snapshot(game: Node2D, tag: String) -> void:
	for i in 20: await process_frame
	await RenderingServer.frame_post_draw
	var target: String = "/tmp/space-idle-native472/nav-cold-switch-" + tag + ".png"
	root.get_texture().get_image().save_png(target)
	var facts: Dictionary = game.session.campaign_system.progress_facts()
	print("SNAP ", tag, " session=", game.session.get_instance_id(), " running=", game.session.run_state.is_running(),
		" pauses=", game.session.pause_reasons, " campaign=", facts.get("state"), " stage=", facts.get("stage"),
		" viewport_update=", game._battle_viewport.render_target_update_mode,
		" viewport_size=", game._battle_viewport.size,
		" surface_visible=", game._battle_viewport.get_parent().is_visible_in_tree(), " path=", target)

func run() -> void:
	root.size = Vector2i(1440, 900)
	root.content_scale_size = Vector2i(1440, 900)
	var game: Node2D = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await _snapshot(game, "normal-equipment")
	var saved: Dictionary = game.session.campaign_system.save_state()
	saved.grants["feature:reactor"] = true
	var granted: Dictionary = game.session.campaign_system.restore_state(saved)
	print("GRANT_RESULT ", granted)
	game._select_left_module("reactor")
	await _snapshot(game, "restored-grant")
	game._select_left_module("equipment")
	await _snapshot(game, "back-equipment")
	game._select_left_module("reactor")
	await _snapshot(game, "warm-reactor")
	quit()
