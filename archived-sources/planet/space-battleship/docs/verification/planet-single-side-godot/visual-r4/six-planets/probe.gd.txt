extends SceneTree
const OUT := "/tmp/planet-six-probe"

func _initialize() -> void:
	call_deferred("run")

func run() -> void:
	DirAccess.make_dir_recursive_absolute(OUT)
	assert(GDExtensionManager.load_extension("/tmp/planet-native-472/planet.gdextension") == GDExtensionManager.LOAD_STATUS_OK)
	var game = load("res://src/main.gd").new()
	root.add_child(game)
	root.position = Vector2i.ZERO
	await process_frame
	assert(game.startup_error.is_empty())
	game.set_process(false)
	game._skip_tutorial = true
	var session = game.session
	var config: Dictionary = JSON.parse_string(FileAccess.get_file_as_string("res://generated/config.json"))
	var campaign: Dictionary = session.campaign_system.save_state()
	for row in config.unlock.unlock:
		campaign.grants[str(row.type) + ":" + str(row.target)] = true
	assert(session.campaign_system.restore_state(campaign).ok)
	var saved: Dictionary = session.planet_system.save_state()
	for n in range(1, 6):
		saved.planets[str(n)].conquered = true
	assert(session.planet_system.restore_state(saved).ok)
	session.planet_system.refresh_availability(session.campaign_system.available)
	game._refresh_feature_visibility()
	game._select_left_module("planet")
	var panel = game.planet_scene
	if game._guidance_panel != null:
		game._guidance_panel.hide()
	for n in range(1, 7):
		assert(session.planet_system.is_available(str(n)))
		panel.select_planet(str(n))
		panel.refresh()
		await create_timer(1.5).timeout
		await process_frame
		await RenderingServer.frame_post_draw
		var name := "planet-%d.png" % n
		var result := root.get_texture().get_image().save_png(OUT.path_join(name))
		assert(result == OK)
		print("SIX_PLANET ", n, " ", panel.selected_planet_id(), " ", name)
	if session.save_files != null:
		session.save_files.close(2000)
	game.queue_free()
	await process_frame
	print("SIX_PLANET_COMPLETE")
	quit()
