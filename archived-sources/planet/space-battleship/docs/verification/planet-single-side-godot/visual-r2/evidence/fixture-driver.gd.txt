extends SceneTree
var game
var session
var panel
var operations: Array = []
const OUT := "/tmp/planet-visual-r2"
func _initialize() -> void:
	call_deferred("run")
func record(name: String, result: Dictionary) -> void:
	operations.append({"operation": name, "result": result})
	print(name, " ", result)
	assert(result.get("ok",false), name)
func shot(name: String) -> void:
	await process_frame
	await process_frame
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT.path_join(name + ".png"))
	var snapshot: Dictionary = session.planet_system.visual_snapshot(panel.selected_planet_id())
	var people: Dictionary = session.crew_system.step_assignments()
	var entry := {"image": name, "selected_planet":panel.selected_planet_id(), "snapshot":snapshot, "facilities":session.planet_system.facility_facts(panel.selected_planet_id(),people.explorers,people.builders), "visible_task":panel._task_button.text, "visible_task_clock":panel._task_line.text, "drawer":panel._open_section, "selected_facility":panel._selected_facility, "facility_entry_pressed":panel._facility_button.button_pressed, "pause_reasons": session.pause_reasons, "game_time": session.game_time, "surface_phase": panel._visual.globe.phases.x, "cloud_phase": panel._visual.globe.phases.y, "visual_clock": panel._visual.phase}
	operations.append(entry)
var popup_evidence: Dictionary = {}
func resolve_picker(kind: String) -> OptionButton:
	return panel.world_planet_picker if kind == "planet" else panel.world_building_picker if kind == "facility" else panel._member_picker if kind == "crew" else panel._assignment_picker

func click(control: Control) -> void:
	var point: Vector2 = control.get_global_rect().get_center() + Vector2(root.position)
	assert(OS.execute("python", ["/tmp/planet-os-input.py", "click", str(point.x), str(point.y)]) == 0)
	await process_frame
	await create_timer(0.14).timeout

func key(name: String) -> void:
	assert(OS.execute("python", ["/tmp/planet-os-input.py", "key", name]) == 0)
	await process_frame
	await create_timer(0.10).timeout

func choose(picker: OptionButton, index: int) -> void:
	var kind := "planet" if picker == panel.world_planet_picker else "facility" if picker == panel.world_building_picker else "crew" if picker == panel._member_picker else "assignment"
	await process_frame
	await process_frame
	picker = resolve_picker(kind)
	var expected = picker.get_item_metadata(index)
	await click(picker)
	if not popup_evidence.has(kind) and kind != "assignment":
		popup_evidence[kind] = true
		await shot("native-" + kind + "-popup")
	assert(picker.get_popup().visible, "native popup did not open")
	var popup := picker.get_popup()
	var point: Vector2 = Vector2(popup.position) + Vector2(float(popup.size.x) * 0.5, float(popup.size.y) * (float(index) + 0.5) / float(popup.get_item_count()))
	operations.append({"native_popup":kind, "window_id":popup.get_window_id(), "position":str(popup.position), "size":str(popup.size), "row":index, "mouse_point":str(point), "method":"XTest OS mouse"})
	assert(OS.execute("python", ["/tmp/planet-os-input.py", "click", str(point.x), str(point.y)]) == 0)
	await create_timer(0.20).timeout
	await process_frame
	var current := resolve_picker(kind)
	assert(current.get_item_metadata(current.selected) == expected, "native choice differs")

func run() -> void:
	DirAccess.make_dir_recursive_absolute(OUT)
	assert(GDExtensionManager.load_extension("/tmp/planet-native-472/planet.gdextension") == GDExtensionManager.LOAD_STATUS_OK)
	game = load("res://src/main.gd").new()
	root.add_child(game)
	root.position = Vector2i.ZERO
	root.grab_focus()
	await process_frame
	print("STARTUP_ERROR ", game.startup_error)
	assert(game.startup_error.is_empty())
	game.set_process(false)
	game._skip_tutorial = true
	session = game.session
	var config: Dictionary = JSON.parse_string(FileAccess.get_file_as_string("res://generated/config.json"))
	var saved: Dictionary = session.campaign_system.save_state()
	for row in config.unlock.unlock: saved.grants[str(row.type)+":"+str(row.target)] = true
	record("synthetic_campaign_grants", session.campaign_system.restore_state(saved))
	record("crew_availability", {"ok": true, "changed":session.crew_system.refresh_availability(session.campaign_system.available)})
	var planets: Dictionary = session.planet_system.save_state()
	var a: Dictionary = planets.planets["1"]
	a.exploration = 240
	a.conquered = true
	for key in a.buildings:
		var row: Dictionary = {}
		for candidate in config.planet_build.planet_build:
			if str(candidate.id) == key: row = candidate
		a.buildings[key].state = "ready" if row.type == "refinery" else "built"
		a.buildings[key].work = int(row.build_explore)
	var b: Dictionary = planets.planets["2"]
	b.exploration = 22
	for key in b.buildings:
		var row: Dictionary = {}
		for candidate in config.planet_build.planet_build:
			if str(candidate.id) == key: row = candidate
		if row.type == "auto_explore":
			b.buildings[key].state = "built"
			b.buildings[key].work = int(row.build_explore)
		elif row.type == "refinery":
			b.buildings[key].state = "building"
			b.buildings[key].work = 2
	record("synthetic_planet_state",session.planet_system.restore_state(planets))
	session.planet_system.refresh_availability(session.campaign_system.available)
	var members: Array = session.crew_system.member_options()
	record("assign_explorer_1",session.coordinate_assignment(str(members[0].id),"planet_explore","1"))
	record("automatic_1",session.planet_system.set_automatic("1",true))
	record("assign_explorer_2",session.coordinate_assignment(str(members[1].id),"planet_explore","2"))
	var started: Dictionary = session.planet_system.save_state()
	started.planets["1"].elapsed = 24.0
	started.planets["2"].elapsed = 8.0
	record("synthetic_elapsed",session.planet_system.restore_state(started))
	session.planet_system.refresh_availability(session.campaign_system.available)
	var contributions: Dictionary = session.planet_system.contributions()
	record("publish_real_fixture_contributions",session.publish_world(contributions))
	game._sync_guidance()
	game._refresh_feature_visibility()
	await click(game._left_buttons.planet)
	panel = game.planet_scene
	panel.select_planet("1")
	panel.refresh()
	await create_timer(3).timeout
	await shot("01-default-planet1")
	await click(panel._facility_button)
	await shot("02-facilities-planet1-ready")
	await choose(panel.world_planet_picker,1)
	await create_timer(2).timeout
	await shot("03-same-drawer-planet2-building")
	assert(panel.selected_planet_id() == "2")
	assert(panel._snapshot.exploration == 22)
	for fact in panel._facts: assert(fact.planet_id == "2")
	assert(panel._drawer.visible and panel._facility_button.button_pressed)
	await choose(panel.world_planet_picker,0)
	await shot("04-switch-back-planet1-ready")
	await click(panel._claim_button)
	await process_frame
	await shot("05-planet1-claimed")
	for fact in panel._facts:
		if fact.id == panel._selected_facility: assert(fact.state == "built")
	await click(panel._stop_button)
	await shot("06-stop-after-current")
	assert(session.planet_system.visual_snapshot("1").running)
	assert(not session.planet_system.visual_snapshot("1").automatic)
	await click(panel._cancel_button)
	await shot("07-cancelled")
	assert(not session.planet_system.visual_snapshot("1").running)
	assert(session.planet_system.visual_snapshot("2").running)
	assert(session.crew_system.step_assignments().explorers.get("1",[]).is_empty())
	assert(session.crew_system.step_assignments().explorers.get("2",[]).size() == 1)
	await click(panel._crew_button)
	await shot("08-crew-section")
	await click(panel._task_button)
	await shot("09-exploration-section")
	await click(panel._crew_button)
	for child in panel._drawer_body.get_children():
		if child is Button and child.text == "派遣":
			await click(child)
			break
	await click(panel._task_button)
	await click(panel._auto_button)
	await click(panel._task_button)
	assert(session.planet_system.visual_snapshot("1").running)
	assert(session.planet_system.visual_snapshot("1").automatic)
	assert(session.crew_system.step_assignments().explorers.get("1",[]).size() == 1)
	game.set_process(true)
	await shot("10-motion-t0-all-built")
	await create_timer(4).timeout
	await shot("11-motion-t4-all-built")
	await create_timer(4).timeout
	await shot("12-motion-t8-all-built")
	await click(panel._facility_button)
	for index in panel._facts.size():
		if panel._facts[index].type == "shipyard":
			await choose(panel.world_building_picker,index)
			break
	await click(panel._reforge_button)
	await process_frame
	assert(game._reforge_planet_value.text == panel.selected_planet_id())
	await shot("13-existing-reforge-context")
	var visible_clip: Rect2 = game.growth_panel.get_global_rect()
	assert(visible_clip.encloses(game._reforge_picker.get_global_rect()))
	operations.append({"visible_reforge_picker_rect":str(game._reforge_picker.get_global_rect()),"visible_growth_rect":str(visible_clip),"reforge_planet_id":game._reforge_planet_value.text})
	game.set_process(false)
	var construction: Dictionary = session.planet_system.save_state()
	construction.planets["2"].exploration = 151
	construction.planets["2"].duration = 3600.0 / 211.0
	construction.planets["2"].elapsed = 8.0
	for row in config.planet_build.planet_build:
		if not construction.planets["2"].buildings.has(str(row.id)): continue
		var build: Dictionary = construction.planets["2"].buildings[str(row.id)]
		build.state = "building" if row.type == "shipyard" else "built"
		build.work = 1 if row.type == "shipyard" else int(row.build_explore)
	record("synthetic_construction_fixture", session.planet_system.restore_state(construction))
	record("publish_construction_fixture_contributions",session.publish_world(session.planet_system.contributions()))
	session.planet_system.publish_status(["2"])
	session.planet_system.buildings_changed.emit(["2"])
	await click(game._left_buttons.planet)
	await choose(panel.world_planet_picker,1)
	await create_timer(1).timeout
	for index in panel._facts.size():
		if panel._facts[index].type == "shipyard":
			await choose(panel.world_building_picker,index)
			break
	await shot("14-missing-builder")
	for fact in panel._facts:
		if fact.type == "shipyard": assert(fact.pause_reason == "MissingBuilders")
	await click(panel._crew_button)
	await choose(panel._member_picker,2)
	for index in panel._assignment_picker.item_count:
		if panel._assignment_picker.get_item_metadata(index).kind == "planet_build":
			await choose(panel._assignment_picker,index)
			break
	for child in panel._drawer_body.get_children():
		if child is Button and child.text == "派遣":
			await click(child)
			break
	await click(panel._facility_button)
	await shot("15-builder-assigned")
	assert(panel._crew_button.text == "人员 2")
	for fact in panel._facts:
		if fact.type == "shipyard":
			assert(fact.pause_reason.is_empty())
			assert(fact.builders_present == 1)
	await click(panel._task_button)
	await click(panel._auto_button)
	await click(panel._facility_button)
	game.set_process(true)
	var waited := 0.0
	var progressed := false
	while waited < 35.0 and not progressed:
		await create_timer(0.5).timeout
		waited += 0.5
		var people: Dictionary = session.crew_system.step_assignments()
		for fact in session.planet_system.facility_facts("2", people.explorers, people.builders):
			if fact.type == "shipyard" and fact.work >= 2: progressed = true
	assert(progressed, "construction failed to progress after builder assignment")
	await shot("16-construction-actually-progressed")
	await click(game._left_buttons.reactor)
	await shot("17-shared-host-reactor")
	await click(game._left_buttons.equipment)
	await shot("18-shared-host-equipment")
	var file := FileAccess.open(OUT.path_join("operations.json"),FileAccess.WRITE)
	file.store_string(JSON.stringify(operations,"  "))
	file.close()
	if session.save_files != null: session.save_files.close(2000)
	game.queue_free()
	await process_frame
	await process_frame
	quit()
