func check(value: bool, label: String) -> void:

func flush() -> void:

func screenshot(name: String) -> void:

func click(control: Control) -> void:
    var point: Vector2 = control.get_global_rect().get_center()
    var down := InputEventMouseButton.new()
    down.position = point; down.global_position = point; down.button_index = MOUSE_BUTTON_LEFT; down.pressed = true
    Input.parse_input_event(down)
    await process_frame
    var up := InputEventMouseButton.new()
    up.position = point; up.global_position = point; up.button_index = MOUSE_BUTTON_LEFT; up.pressed = false
    Input.parse_input_event(up)
    await flush()
func snapshot() -> Dictionary:
    return {"wallet": host.economy.budget().duplicate(true), "wallet_text": host.economy.balance_rows(), "slots": host.equipment_slot_facts(), "native": host.battle.hud_values(), "game_time": host.game_time, "pause_reasons": host.pause_reasons.duplicate(), "candidate": panel._candidate, "selected_slot": panel._selected, "comparison": panel.last_comparison.duplicate(true), "quantity": panel._quantity, "maximum": panel._maximum}
func record(action: String, before: Dictionary) -> void:
    records.append({"action": action, "before": before, "after": snapshot(), "actual_result": panel.last_result.duplicate(true)})
func assert_purchase(before: Dictionary, label: String) -> void:
    var result: Dictionary = panel.last_result
    check(result.get("purchase_receipt", {}).get("commit_status") == "accepted", label + " native accepted receipt")
    var after: Dictionary = snapshot()
    var spent: Dictionary = result.get("purchase_receipt", {}).get("spent", {})
    for resource in spent:
        var expected: Dictionary = host.amount_api.apply("sub_nonnegative", [before.wallet[resource], spent[resource]])
        check(expected.ok and host.amount_api.apply("compare", [expected.value, after.wallet[resource]]).scalar == 0, label + " exact wallet minus actual spent " + resource)
    for purchase in result.get("purchase_receipt", {}).get("purchases", []):
        var old_level: int = -1
        var new_level: int = -1
        for row in before.slots:
            if row.id == purchase.slot_id: old_level = row.paid_level
        for row in after.slots:
            if row.id == purchase.slot_id: new_level = row.paid_level
        check(new_level - old_level == int(purchase.count), label + " actual count " + purchase.slot_id)
func wallet(iron: String, material: String) -> void:
    var state: Dictionary = host.economy.save_state()
    state.wallet = {"1": host.amount_api.apply("from_decimal", [iron]).value, "2": host.amount_api.apply("from_decimal", ["1000"]).value, "degenerate_matter": host.amount_api.apply("from_decimal", [material]).value}
    check(host.economy.restore_state(state).ok, "validated controlled wallet setup")
    host.economy.publish_budget_change()
func nodes_of(node: Node, nodes: Dictionary, styles: Dictionary, texts: Dictionary) -> void:
    nodes[str(node.get_instance_id())] = str(node.get_path())
    if node is Button:
        styles[str(node.get_instance_id())] = node.get_theme_stylebox("normal").get_instance_id()
    if node is Label: texts[str(node.get_instance_id())] = node.text
    for child in node.get_children(): nodes_of(child, nodes, styles, texts)
func perf_snapshot() -> Dictionary:
    var nodes: Dictionary = {}; var styles: Dictionary = {}; var texts: Dictionary = {}
    nodes_of(app, nodes, styles, texts)
    var resources: Dictionary = {}
    for key in host.battle._models: resources[key] = host.battle._models[key].texture.get_instance_id()
    return {"nodes":nodes,"styles":styles,"texts":texts,"observed":observed.duplicate(),"configuration_commits":host.battle.configuration_commits,

        "hull_resources":resources,"weapons":host.battle._values.duplicate(true),"enemies":host.battle._kernel.enemy_views(),"workload":host.battle._kernel.workload_metrics(),"persistent":host.battle._kernel.persistent_state(),"frames":Engine.get_frames_drawn(),"equipment_counters":panel.get("presentation_counts").duplicate(true) if panel.get("presentation_counts") != null else null}
func perf_record(label: String, before: Dictionary) -> void:
    var after: Dictionary = perf_snapshot()
    var added: Array = []; var removed: Array = []; var styles: Array = [];var texts: Array = []
    for key in after.nodes:
        if not before.nodes.has(key): added.append(after.nodes[key])
    for key in before.nodes:
        if not after.nodes.has(key): removed.append(before.nodes[key])
    for key in after.styles:
        if before.styles.has(key) and before.styles[key] != after.styles[key]: styles.append(after.nodes[key])
    for key in after.texts:
        if before.texts.has(key) and before.texts[key] != after.texts[key]: texts.append(after.nodes[key])
    var delta: Dictionary = {}
    for key in observed: delta[key] = after.observed[key] - before.observed[key]
    var differences: Dictionary = {"weapons":[],"defences":[]}
    var ui_delta: Dictionary = {}
    if before.equipment_counters != null:
        for key in after.equipment_counters: ui_delta[key] = after.equipment_counters[key] - before.equipment_counters[key]
    for group in differences:
        for key in after.weapons[group]:
            if after.weapons[group][key] != before.weapons[group].get(key): differences[group].append(key)
    var stable: bool = before.kernel_id == after.kernel_id and before.presentation_id == after.presentation_id and before.statistics_id == after.statistics_id and before.hull_resources == after.hull_resources
    perf_records.append({"action":label,"before":before,"after":after,"observed_delta":delta,"added_nodes":added,"removed_nodes":removed,"replaced_button_styles":styles,"changed_label_texts":texts,"changed_battle_values":differences,"equipment_counter_delta":ui_delta,"battle_objects_and_resources_same":stable,"same_enemy_array_identity":is_same(before.enemies,after.enemies),"configuration_commit_delta":after.configuration_commits-before.configuration_commits,"result":panel.last_result.duplicate(true)})
    check(stable,"stable battle/resources "+label)
    check(before.enemies == after.enemies,"unrelated enemies unchanged "+label)
    for key in ["active_flights","flight_slots","active_capacity","feedback_capacity"]:
        check(before.workload[key] == after.workload[key],"flight pool retained "+key+" "+label)
    if label != "actual_refit_cannon": check(before.workload == after.workload,"queues retained "+label)
    check(after.configuration_commits-before.configuration_commits == (1 if label=="actual_refit_cannon" else 0), "configuration commit scope "+label)
    if label != "actual_refit_cannon":
        check(before.persistent == after.persistent,"encounter/flight serial and history retained "+label)
    else:
        check(before.persistent.history_counts == after.persistent.history_counts and before.persistent.next_ids.attack == after.persistent.next_ids.attack and before.persistent.next_ids.encounter == after.persistent.next_ids.encounter, "refit retains in-flight attack/encounter identity")
func perf_click(label: String, control: Control) -> void:

    perf_record(label,before)
