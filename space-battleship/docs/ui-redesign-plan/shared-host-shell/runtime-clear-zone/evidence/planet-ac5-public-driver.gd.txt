extends SceneTree
const OUT := "res://docs/ui-redesign-plan/shared-host-shell/runtime-clear-zone/evidence/"
func _initialize() -> void: call_deferred("run")
func frames(count: int) -> void:
	for i in count: await process_frame
func shot(name: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT + name)
func run() -> void:
	root.size = Vector2i(1440, 900)
	root.content_scale_size = Vector2i(1440, 900)
	var game = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await frames(90)
	assert(game.startup_error.is_empty())
	game.set_process(false)
	game._skip_tutorial = true
	var session = game.session
	var config: Dictionary = JSON.parse_string(FileAccess.get_file_as_string("res://generated/config.json"))
	var saved: Dictionary = session.campaign_system.save_state()
	for row in config.unlock.unlock: saved.grants[str(row.type) + ":" + str(row.target)] = true
	assert(session.campaign_system.restore_state(saved).ok)
	session.crew_system.refresh_availability(session.campaign_system.available)
	session.reactor_system.refresh_availability(session.campaign_system.available)
	assert(session.reactor_system.refresh(session.reactor_inputs(), session.publish_reactor).ok)
	var planets: Dictionary = session.planet_system.save_state().duplicate(true)
	for id in planets.planets:
		var state: Dictionary = planets.planets[id]
		state.exploration = 240
		state.conquered = true
		for key in state.buildings:
			state.buildings[key].state = "built"
			for candidate in config.planet_build.planet_build:
				if str(candidate.id) == key: state.buildings[key].work = int(candidate.build_explore)
	assert(session.planet_system.restore_state(planets).ok)
	session.planet_system.refresh_availability(session.campaign_system.available)
	game._sync_guidance()
	game._refresh_feature_visibility()
	assert(game._zone_shell.is_route_available("planet"))
	game._zone_shell._route_buttons["planet"].pressed.emit()
	await frames(60)
	assert(game.planet_scene.visible and not game.equipment_scene.visible and not game.reactor_scene.visible)
	assert(game._zone_shell._active_route == "planet")
	assert(not game.planet_scene._wallet.visible)
	game.planet_scene.select_planet("1")
	await frames(60)
	await shot("planet-ac5-public-stage4-1440x900.png")
	print("PLANET_AC5 route=", game._zone_shell._active_route, " selected=", game.planet_scene.selected_planet_id(), " scene=", game.planet_scene.size, " wallet_hidden=", not game.planet_scene._wallet.visible)
	game.planet_scene._facility_button.pressed.emit()
	game.planet_scene._selected_facility = "workshop"
	game.planet_scene._rebuild_drawer()
	await frames(8)
	assert(game.planet_scene._open_section == "facilities")
	await shot("planet-ac5-public-facilities-1440x900.png")
	game.planet_scene._task_button.pressed.emit()
	await frames(8)
	assert(game.planet_scene._open_section == "exploration")
	var conquest_opened := false
	for child in game.planet_scene._drawer_body.get_children():
		if child is Button and child.text.begins_with("征服奖励"):
			child.pressed.emit()
			conquest_opened = true
			break
	assert(conquest_opened and game.planet_scene._conquest_open)
	await frames(8)
	await shot("planet-ac5-public-conquest-1440x900.png")
	game.planet_scene._toggle_section("exploration")
	game.planet_scene.select_planet("3")
	await frames(90)
	assert(game.planet_scene.selected_planet_id() == "3" and game.planet_scene._visual.planet_id == "3" and not game.planet_scene._visual.preparing)
	await shot("planet-ac5-public-main-star-1440x900.png")
	game.planet_scene.select_planet("4")
	await frames(90)
	assert(game.planet_scene.selected_planet_id() == "4" and game.planet_scene._visual.planet_id == "4" and not game.planet_scene._visual.preparing)
	await shot("planet-ac5-public-red-giant-1440x900.png")
	game.planet_scene.world_planet_picker.grab_focus()
	game._zone_shell._route_buttons["equipment"].pressed.emit()
	await frames(5)
	var focus: Control = root.gui_get_focus_owner()
	assert(game.equipment_scene.visible and not game.planet_scene.visible)
	assert(focus == null or not game.planet_scene.is_ancestor_of(focus))
	await shot("planet-ac5-public-equipment-1440x900.png")
	game._zone_shell._route_buttons["reactor"].pressed.emit()
	await frames(8)
	assert(game.reactor_scene.visible and not game.planet_scene.visible and not game.equipment_scene.visible)
	await shot("planet-ac5-public-reactor-1440x900.png")
	game._zone_shell._route_buttons["universe"].pressed.emit()
	await frames(5)
	assert(game.growth_panel.visible and not game.planet_scene.visible and game._zone_shell._active_route == "universe")
	await shot("planet-ac5-public-universe-1440x900.png")
	print("PLANET_AC5 return_equipment_and_universe=true focus_clear=true")
	if session.save_files != null: session.save_files.close(2000)
	game.queue_free()
	await frames(3)
	quit()
