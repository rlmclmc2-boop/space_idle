const W := Color("e9eee6")

class Study:

    var variant: String
    var font: Font
    var screen_e: Texture2D

    var surface_tex: Texture2D
    func _init(v: String, f: Font, e: Texture2D, r: Texture2D, m: Texture2D) -> void:
        variant = v; font = f; screen_e = e; screen_r = r; surface_tex = m
    func label(value: String, x: float, y: float, size: int, color: Color = Color("e9eee6")) -> void:
        draw_string(font, Vector2(x,y), value, HORIZONTAL_ALIGNMENT_LEFT, -1, size, color)
    func box(r: Rect2, fill: Color, stroke: Color, width: float = 1.0) -> void:
        draw_rect(r, fill); draw_rect(r, stroke, false, width)
    func button(r: Rect2, value: String, primary: bool = false, disabled: bool = false) -> void:
        var body := Color("294d58") if not disabled else Color("213943")
        box(r, body, GOLD if primary else Color("779999"), 1.5)
        draw_line(r.position + Vector2(3,2), r.position + Vector2(r.size.x-3,2), Color("88acab") if not disabled else Color("527277"), 1)
        draw_line(r.position + Vector2(2,r.size.y-3), r.position + Vector2(r.size.x-2,r.size.y-3), Color("0b2733"), 2)
        var tx := r.position.x + maxf(5,(r.size.x-font.get_string_size(value,HORIZONTAL_ALIGNMENT_LEFT,-1,14).x)/2)
        label(value,tx,r.position.y+r.size.y/2+5,14,Color("f8dfac") if primary else W if not disabled else Color("9aaba8"))
    func state_button(r: Rect2, caption: String, state: String) -> void:
        var fill := Color("294d58")
        var stroke := Color("779999")
        var text_color := W
        if state == "hover": fill = Color("3b6470"); stroke = Color("a4c6bc")
        elif state == "pressed": fill = Color("183842"); stroke = Color("779999")
        elif state == "selected": fill = Color("345e68"); stroke = Color("77b7bd")
        elif state == "disabled": fill = Color("263e47"); stroke = Color("49666c"); text_color = Color("91a8a7")
        elif state == "affordable": fill = Color("294d58"); stroke = GOLD; text_color = Color("ffe0ad")
        box(r,fill,stroke,2 if state == "selected" else 1)
        if state == "focus": draw_rect(r.grow(3),Color("a8d4d1"),false,1)
        if state == "pressed": draw_line(r.position+Vector2(3,r.size.y-3),r.end-Vector2(3,3),Color("091f2b"),3)
        label(caption,r.position.x+10,r.position.y+30,15,text_color)
    func base_panel(r: Rect2) -> void:
        draw_texture_rect_region(surface_tex,r,Rect2(430,230,812,155))
        draw_rect(r,Color("3d6972"),false,2)
        draw_line(Vector2(r.position.x+12,r.position.y+3),Vector2(r.end.x-12,r.position.y+3),Color("b4ccbb",0.75),1)
        draw_line(Vector2(r.position.x+12,r.end.y-4),Vector2(r.end.x-12,r.end.y-4),Color("0a2730"),3)
    func ivory(r: Rect2) -> void:
        draw_texture_rect_region(surface_tex,r,Rect2(450,550,770,105))
        draw_rect(r,Color("819c9b"),false,1.5)
        draw_line(r.position+Vector2(4,2),Vector2(r.end.x-4,r.position.y+2),Color("ffffff",0.6),1)

            label("⌄",589,top+27,18,Color("507982"))
            draw_line(Vector2(394,top+31),Vector2(604,top+31),Color("89a9a7"),1)

            label("装备 d354 / 19a2    ·    反应堆 6d / 808",1754,870,13,MUTED)

func texture(path: String) -> Texture2D:
    var image := Image.new()
    if image.load(path) != OK: push_error("Missing source: "+path); return null
    return ImageTexture.create_from_image(image)
func _initialize() -> void:
    call_deferred("run")
func run() -> void:
    var font := FontVariation.new()
    font.base_font = load("res://assets/fonts/NotoSansSC.ttf")
    font.variation_opentype = {2003265652:500.0}
    var e := texture(OUT+"source-equipment-19a2-early2.png")
    var r := texture(ORIGINAL+"r51c6d-combo-reactor-1440x900.png")
    var material := texture(OUT+"material-reference.png")
    for v in ["equipment","reactor","comparison"]:
        root.size = Vector2i(2400,1030) if v=="comparison" else Vector2i(1440,900)
        root.content_scale_size = root.size
        var study := Study.new(v,font,e,r,material)
        root.add_child(study)
        study.size = root.size
        await process_frame
        await RenderingServer.frame_post_draw
        root.get_texture().get_image().save_png(OUT + ("compare-2400x1030.png" if v=="comparison" else v+"-1440x900.png"))
        study.queue_free()
        await process_frame
    quit()
