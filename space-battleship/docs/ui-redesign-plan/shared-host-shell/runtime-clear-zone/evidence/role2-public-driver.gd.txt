extends SceneTree

func _initialize() -> void:
	call_deferred("run")

func frames(n: int) -> void:
	for i in n: await process_frame

func clear_guidance(session, game) -> void:
	var pending: Dictionary = session.pending_guidance()
	var tries := 0
	while not pending.is_empty() and tries < 20:
		assert(session.dismiss_guidance(str(pending.kind), str(pending.id)).ok)
		pending = session.pending_guidance()
		tries += 1
	game._sync_guidance()

func shot(name: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://docs/ui-redesign-plan/shared-host-shell/runtime-clear-zone/evidence/" + name + "-1440x900.png")

func run() -> void:
	root.size = Vector2i(1440, 900)
	root.content_scale_size = Vector2i(1440, 900)
	var game: Node2D = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await frames(45)
	assert(game.session != null and game.startup_error.is_empty())
	clear_guidance(game.session, game)
	game._refresh_feature_visibility()
	game._zone_shell._route_buttons["settings"].pressed.emit()
	await frames(6)
	clear_guidance(game.session, game)
	assert(game._music_toggle != null and game._effects_toggle != null)
	await frames(3)
	await shot("role2-public-settings-normal")
	Input.warp_mouse(game._music_toggle.get_global_rect().get_center())
	await frames(3)
	await shot("role2-public-settings-hover")
	game._music_toggle.grab_focus()
	await frames(2)
	await shot("role2-public-settings-focus")
	game._music_toggle.release_focus()
	game._effects_toggle.disabled = true
	await frames(2)
	await shot("role2-public-settings-disabled")
	game._effects_toggle.disabled = false
	game._zone_shell._show_balances()
	await frames(4)
	await shot("role2-public-resources")
	print("ROLE2_PUBLIC settings_and_resources captured; music_checked=", game._music_toggle.button_pressed)
	if game.session.save_files != null: game.session.save_files.close(2000)
	game.queue_free()
	await frames(3)
	quit()
