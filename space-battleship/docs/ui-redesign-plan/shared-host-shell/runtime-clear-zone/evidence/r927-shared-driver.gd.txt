extends SceneTree

func _initialize() -> void:
	call_deferred("run")

func frames(n: int) -> void:
	for i in n: await process_frame

func clear_guidance(session, game) -> void:
	var pending: Dictionary = session.pending_guidance()
	var tries := 0
	while not pending.is_empty() and tries < 20:
		assert(session.dismiss_guidance(str(pending.kind), str(pending.id)).ok)
		pending = session.pending_guidance()
		tries += 1
	game._sync_guidance()

func run() -> void:
	root.size = Vector2i(1440, 900)
	root.content_scale_size = Vector2i(1440, 900)
	var game: Node2D = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await frames(45)
	var session = game.session
	assert(session != null and game.startup_error.is_empty())
	assert(session.current_chrono_speed() == 1)
	var saved: Dictionary = session.campaign_system.save_state()
	saved.grants["feature:reactor"] = true
	saved.grants["reactor_module:condensation"] = true
	assert(session.campaign_system.restore_state(saved).ok)
	session.reactor_system.refresh_availability(session.campaign_system.available)
	assert(session.reactor_system.refresh(session.reactor_inputs(), session.publish_reactor).ok)
	clear_guidance(session, game)
	game._refresh_feature_visibility()
	game._zone_shell._route_buttons["reactor"].pressed.emit()
	await frames(5)
	var scene = game.reactor_scene
	assert(scene != null and scene.visible)
	assert(is_zero_approx(float(session.reactor_system.presentation_model().load)))
	clear_guidance(session, game)
	await frames(2)
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://docs/ui-redesign-plan/shared-host-shell/runtime-clear-zone/evidence/r927-shared-zero-1440x900.png")
	var amount: Dictionary = session.amount_api.apply("from_decimal", ["1000"])
	assert(amount.ok)
	assert(session.economy.credit_yield("2", amount.value, session.wall_time, session.game_time).ok)
	await frames(3)
	scene._equal.pressed.emit()
	await frames(5)
	clear_guidance(session, game)
	var model: Dictionary = session.reactor_system.presentation_model()
	assert(scene.last_result.get("ok", false))
	assert(float(model.load) > 0.0)
	for row in model.modules:
		assert(float(row.load) > 0.0)
	# Let the native command notice expire before the continuous capture begins.
	await create_timer(2.3).timeout
	clear_guidance(session, game)
	await frames(2)
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://docs/ui-redesign-plan/shared-host-shell/runtime-clear-zone/evidence/r927-shared-active-1440x900.png")
	assert(scene._quote.ok and scene._quote.can_purchase)
	scene._quantity.pressed.emit()
	await frames(3)
	assert(scene._quantity.text == "×10" and scene._quote.ok and int(scene._quote.count) == 10)
	print("R927_SHARED TEN quote=", scene._quote, " level=", session.reactor_system.presentation_model().level)
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://docs/ui-redesign-plan/shared-host-shell/runtime-clear-zone/evidence/r927-shared-ten-1440x900.png")
	scene._quantity.pressed.emit()
	await frames(8)
	assert(scene._quantity.text == "MAX" and scene._quote.ok and int(scene._quote.count) >= 0)
	assert(scene._cost_badge.visible and scene._cost_icon.texture != null)
	print("R927_SHARED MAX quote=", scene._quote, " level=", session.reactor_system.presentation_model().level, " badge=", scene._cost_badge.visible)
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://docs/ui-redesign-plan/shared-host-shell/runtime-clear-zone/evidence/r927-shared-max-1440x900.png")
	game._zone_shell._route_buttons["equipment"].pressed.emit()
	await frames(5)
	assert(game.equipment_scene.visible and not game.reactor_scene.visible)
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://docs/ui-redesign-plan/shared-host-shell/runtime-clear-zone/evidence/r927-shared-equipment-1440x900.png")
	print("R927_SHARED routes=reactor_equipment scene=808x900")
	if session.save_files != null: session.save_files.close(2000)
	game.queue_free()
	await frames(3)
	quit()
