class Study:
    extends Control
    var variant: String
    var font: Font
    var equipment_source: Texture2D
    var equipment_late: Texture2D
    var reactor_source: Texture2D
    var controls: Texture2D
    var grabber: Texture2D
    func _init(v: String, f: Font, e: Texture2D, el: Texture2D, r: Texture2D, c: Texture2D, g: Texture2D) -> void:
        variant=v; font=f; equipment_source=e; equipment_late=el; reactor_source=r; controls=c; grabber=g
    func text(s:String,x:float,y:float,size:int,color:Color=CREAM) -> void:
        draw_string(font,Vector2(x,y),s,HORIZONTAL_ALIGNMENT_LEFT,-1,size,color)
    func line(a:Vector2,b:Vector2,color:Color,width:float=1.0) -> void:
        draw_line(a,b,color,width)
    func frame(r:Rect2,fill:Color,border:Color,width:float=1.0) -> void:
        draw_rect(r,fill);draw_rect(r,border,false,width)
    func nine(src:Rect2,dest:Rect2,tint:Color=Color.WHITE) -> void:
        var sx:float=65.0;var sy:float=65.0;var dx:float=minf(13.0,dest.size.x*.16);var dy:float=minf(10.0,dest.size.y*.25)
        var sw:float=src.size.x-2*sx;var sh:float=src.size.y-2*sy
        var dw:float=dest.size.x-2*dx;var dh:float=dest.size.y-2*dy
        var ws:=[sx,sw,sx];var hs:=[sy,sh,sy];var wd:=[dx,dw,dx];var hd:=[dy,dh,dy]
        var yy:float=src.position.y;var ty:float=dest.position.y
        for j in 3:
            var xx:float=src.position.x;var tx:float=dest.position.x
            for i in 3:
                draw_texture_rect_region(controls,Rect2(tx,ty,wd[i],hd[j]),Rect2(xx,yy,ws[i],hs[j]),tint)
                xx+=ws[i];tx+=wd[i]
            yy+=hs[j];ty+=hd[j]
    func plate(r:Rect2) -> void:
        nine(Rect2(45,131,1360,250),r)
    func key(r:Rect2,s:String,primary:bool=false,state:String="normal") -> void:
        var src:=Rect2(45,470,1360,220) if primary else Rect2(45,770,1360,210)
        var tint:=Color.WHITE
        if state=="hover":tint=Color(1.13,1.16,1.14)
        elif state=="pressed":tint=Color(.78,.84,.84)
        elif state=="disabled":tint=Color(.55,.62,.65,.85)
        nine(src,r,tint)
        if state=="selected":draw_rect(r.grow(1),Color("72bdc0"),false,2)
        if state=="focus":draw_rect(r.grow(3),Color("b9e5d8"),false,1.5)
        var color:=INK if primary else CREAM
        if state=="disabled":color=Color("aab6b2")
        var x:=r.position.x+maxf(4.0,(r.size.x-font.get_string_size(s,HORIZONTAL_ALIGNMENT_LEFT,-1,15).x)/2.0)
        text(s,x,r.position.y+r.size.y/2+5,15,color)
    func select_field(r:Rect2,name:String) -> void:
        # A subdued instance of the same lacquer cross-section as the action
        # keys. The arrow is a drawn grab point, not a tiny font glyph.
        nine(Rect2(45,770,1360,210),r,Color(.79,.94,.95))
        text(name,r.position.x+12,r.position.y+r.size.y/2+5,15,CREAM)
        var x:=r.end.x-23.0;var y:=r.position.y+r.size.y/2-2.0
        draw_colored_polygon(PackedVector2Array([Vector2(x,y),Vector2(x+14,y),Vector2(x+7,y+7)]),Color("e8e0c6"))
    func equipment_row(source:Texture2D,top:float,icon_src:Rect2,name:String,level:String,stat:String,cost:String,mode:String,buy:String,poor:bool=false) -> void:
        plate(Rect2(340,top,432,91))
        draw_texture_rect_region(source,Rect2(348,top+22,42,48),icon_src)
        select_field(Rect2(399,top+6,218,27),name)
        text(level,630,top+26,14,Color("45636b"))
        key(Rect2(704,top+6,65,28),"比较")
        var stat_parts:=stat.split("  ",false,1)
        text(stat_parts[0],407,top+55,15,INK)
        text(stat_parts[1] if stat_parts.size()>1 else "",470,top+55,15,Color("26718a"))
        text("费用",407,top+80,15,INK)
        draw_texture_rect_region(source,Rect2(441,top+63,20,20),Rect2(445,top+63,20,20))
        if mode=="MAX":
            text("铁",465,top+80,15,INK)
            text("1.1e39",484,top+80,15,Color("26718a"))
            draw_texture_rect_region(source,Rect2(537,top+64,19,19),Rect2(534,top+64,19,19))
            text("铀",560,top+80,15,INK)
            text("2.8k",579,top+80,15,Color("a33725"))
        else:
            text("铁",465,top+80,15,INK)
            text("65.0",484,top+80,15,Color("a33725") if poor else Color("26718a"))
        key(Rect2(616,top+60,50,28),mode)
        key(Rect2(671,top+60,98,28),buy,true)
    func equipment_page(late:bool=false) -> void:
        var source:=equipment_late if late else equipment_source
        draw_texture_rect(source,Rect2(0,0,1440,900),false)
        # The latest source already has the correct real carrier, ship size,
        # row viewport and host. Replace only its existing inner face.
        var inner:=Rect2(339,260,434,522) if late else Rect2(339,399,434,278)
        draw_rect(inner,Color("12333f"))
        frame(Rect2(inner.position.x+2,inner.position.y,430,36),Color("183e4a"),Color("729b9b"),1)
        key(Rect2(677,inner.position.y+4,94,29),"全部升级")
        if late:
            text("武器",350,307,13,CREAM)
            var names:=["脉冲炮","导弹","舰炮","长激光","脉冲炮"]
            var levels:=["Lv.19","Lv.1","Lv.1","Lv.1","Lv.1"]
            var stats:=["攻击力  9.3k → 1.6e43","攻击力  170.0 → 200.0","攻击力  1.2k → 1.4k","攻击力  35.0 → 45.0","攻击力  350.0 → 420.0"]
            var icons:=[Rect2(349,337,42,48),Rect2(349,432,42,48),Rect2(349,526,42,48),Rect2(349,620,42,48),Rect2(349,714,42,48)]
            for i in 5:
                var top:=314.0+i*94.0
                equipment_row(source,top,icons[i],names[i],levels[i],stats[i],"铁65.0","MAX" if i==0 else "×1","购买444级" if i==0 else "购买1级")
            draw_rect(Rect2(340,782,432,18),Color("12333f"))
            text("防御",350,797,13,CREAM)
        else:
            text("武器",350,448,13,CREAM)
            equipment_row(source,452,Rect2(349,473,42,48),"脉冲炮","Lv.1","攻击力  350.0 → 420.0","铁65.0","×1","购买1级",true)
            text("防御",350,561,13,CREAM)
            equipment_row(source,565,Rect2(349,586,42,48),"装甲","Lv.1","容量  1.0k → 1.2k","铁65.0","×1","购买1级",true)
    func reactor_page() -> void:
        draw_texture_rect(reactor_source,Rect2(0,0,1440,900),false)

        # against the stage. Cool their warm metal without erasing geometry.

        nine(Rect2(45,770,1360,210),Rect2(75,548,563,53),Color(.83,.91,.91))

        key(Rect2(652,555,94,38),"均分")
        plate(Rect2(74,607,492,225))
        plate(Rect2(572,607,207,225))

        var names:=["武器","盾甲","铁产出","碎片"]
        for i in 4:
            var y:float=611+i*54.0
            if i>0:line(Vector2(84,y),Vector2(554,y),Color("9bb1ae"),1)
            draw_rect(Rect2(83,y+6,4,38),colors[i])
            text(names[i],99,y+34,17,INK)
            frame(Rect2(198,y+17,154,18),Color("173d4c"),Color("5e7f81"),1)
            draw_rect(Rect2(201,y+20,37,12),colors[i])

            text("×1.13",479,y+34,15,Color("3b6068"))

        line(Vector2(591,670),Vector2(758,670),Color("9db1ad"))
        text("费用",592,704,14,INK)
        draw_texture_rect_region(reactor_source,Rect2(631,687,21,21),Rect2(657,687,23,23))

func texture(path:String) -> Texture2D:
    var image:=Image.new()
    if image.load(path)!=OK:push_error("Missing "+path);return null
    return ImageTexture.create_from_image(image)
func _initialize() -> void:
    call_deferred("run")
func run() -> void:
    var font:=FontVariation.new()
    font.base_font=load("res://assets/fonts/NotoSansSC.ttf")
    font.variation_opentype={2003265652:500.0}
    var e:=texture(EQUIP_SOURCE);var el:=texture(EQUIP_LATE);var r:=texture(REACTOR_SOURCE)
    var c:=texture("res://assets/ui/equipment/materials-20261029/control-materials.png")
    var g:=texture("res://assets/ui/reactor/b01/slider-knob.png")
    for v in ["equipment","equipment-late","reactor","comparison"]:
        root.size=Vector2i(2400,1030) if v=="comparison" else Vector2i(1440,900)
        root.content_scale_size=root.size
        var study:=Study.new(v,font,e,el,r,c,g)
        root.add_child(study);study.size=root.size
        await process_frame;await RenderingServer.frame_post_draw
        root.get_texture().get_image().save_png(OUT+("compare-2400x1030.png" if v=="comparison" else v+"-1440x900.png"))
        study.queue_free();await process_frame
    quit()
