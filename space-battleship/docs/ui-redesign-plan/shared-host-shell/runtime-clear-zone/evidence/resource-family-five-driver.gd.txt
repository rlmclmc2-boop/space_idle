extends SceneTree
const OUT := "res://docs/ui-redesign-plan/shared-host-shell/runtime-clear-zone/evidence/"
func _initialize() -> void:
	call_deferred("run")
func frames(count: int) -> void:
	for i in count: await process_frame
func shot(name: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT + name)
func run() -> void:
	root.size = Vector2i(1440, 900)
	root.content_scale_size = Vector2i(1440, 900)
	var game = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await frames(90)
	if not game.startup_error.is_empty():
		print("INTEGRATION startup_failed ", game.startup_error)
		quit(1)
		return
	# Controlled UI-state observation: credit one unit through the public Economy API
	# so the otherwise hidden fifth resource row becomes visible. Not natural progress.
	var matter_amount: Dictionary = game.session.amount_api.apply("from_decimal", ["1"])
	assert(matter_amount.ok)
	var credited: Dictionary = game.session.economy.credit_yield("degenerate_matter", matter_amount.value, game.session.wall_time, game.session.game_time)
	assert(credited.ok)
	game._skip_tutorial = true
	game._sync_guidance()
	# The saved page preference can reopen the ship page; select equipment explicitly.
	game._zone_shell._route_buttons["equipment"].pressed.emit()
	await frames(4)
	assert(game._zone_shell._active_route == "equipment")
	assert(game._zone_shell._context.text == "装备")
	assert(game._zone_shell._route_buttons["equipment"].button_pressed)
	assert(game.equipment_scene.visible and not game.growth_panel.visible and not game.reactor_scene.visible)
	game._zone_shell._show_balances()
	await frames(4)
	await shot("resource-family-five-1440x900.png")
	print("EQUIPMENT_CAPTURE route=", game._zone_shell._active_route, " title=", game._zone_shell._context.text, " nav=", game._zone_shell._route_buttons["equipment"].button_pressed, " panel=", game.equipment_scene.visible)
	quit()
