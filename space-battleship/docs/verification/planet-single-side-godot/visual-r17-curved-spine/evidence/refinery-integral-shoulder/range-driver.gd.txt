extends SceneTree
const OUT := "/tmp/planet-refinery-candidate-range"
func _initialize() -> void: call_deferred("run")
func shot(index: int) -> void:
	await process_frame
	await process_frame
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT.path_join("frame_%03d.png" % index))
func run() -> void:
	DirAccess.make_dir_recursive_absolute(OUT)
	root.size = Vector2i(1440, 900)
	var backdrop := ColorRect.new()
	backdrop.color = Color("061723")
	backdrop.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	root.add_child(backdrop)
	var visual = load("res://scripts/planet_visual.gd").new()
	backdrop.add_child(visual)
	visual.position = Vector2(400, 45)
	visual.size = Vector2(808, 810)
	await process_frame
	visual.globe.prepare_material()
	var entries: Array = []
	var types := ["auto_explore", "refinery", "equipment", "shipyard"]
	for index in 4: entries.append({"id":str(index), "type":types[index], "owned":true})
	visual.configure("1", {"visual": {}, "visualFacilities":entries})
	visual.set_exploration(true, 240, 20.0)
	for index in 97:
		if index > 0: visual.advance(TAU / (96.0 * 0.045), false)
		if index in [22, 24, 26, 49, 51, 53]: await shot(index)
	quit()
