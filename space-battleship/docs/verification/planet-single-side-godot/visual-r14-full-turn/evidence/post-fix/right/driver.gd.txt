var operations: Array = []
const OUT := "/tmp/planet-r14-horizon-art-fix-r3-right"
func _initialize() -> void:
	call_deferred("run")
func record(name: String, result: Dictionary) -> void:
	operations.append({"operation": name, "result": result})
	print(name, " ", result)
	assert(result.get("ok",false), name)
func shot(name: String) -> void:
	await process_frame
	await process_frame
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT.path_join(name + ".png"))
	var snapshot: Dictionary = session.planet_system.visual_snapshot(panel.selected_planet_id())
	var people: Dictionary = session.crew_system.step_assignments()
	var entry := {"image": name, "selected_planet":panel.selected_planet_id(), "snapshot":snapshot, "facilities":session.planet_system.facility_facts(panel.selected_planet_id(),people.explorers,people.builders), "visible_task":panel._task_button.text, "visible_task_clock":panel._task_line.text, "drawer":panel._open_section, "selected_facility":panel._selected_facility, "facility_entry_pressed":panel._facility_button.button_pressed, "pause_reasons": session.pause_reasons, "game_time": session.game_time, "surface_phase": panel._visual.globe.phases.x, "cloud_phase": panel._visual.globe.phases.y, "visual_clock": panel._visual.phase}
	operations.append(entry)
var popup_evidence: Dictionary = {}
func resolve_picker(kind: String) -> OptionButton:
	return panel.world_planet_picker if kind == "planet" else panel.world_building_picker if kind == "facility" else panel._member_picker if kind == "crew" else panel._assignment_picker

func click(control: Control) -> void:
	var point: Vector2 = control.get_global_rect().get_center() + Vector2(root.position)
	assert(OS.execute("python", ["/tmp/planet-os-input.py", "click", str(point.x), str(point.y)]) == 0)
	await process_frame
	await create_timer(0.14).timeout

func key(name: String) -> void:
	assert(OS.execute("python", ["/tmp/planet-os-input.py", "key", name]) == 0)
	await process_frame
	await create_timer(0.10).timeout

func choose(picker: OptionButton, index: int) -> void:
	var kind := "planet" if picker == panel.world_planet_picker else "facility" if picker == panel.world_building_picker else "crew" if picker == panel._member_picker else "assignment"
	await process_frame
	await process_frame
	picker = resolve_picker(kind)
	var expected = picker.get_item_metadata(index)
	await click(picker)
	if not popup_evidence.has(kind) and kind != "assignment":
		popup_evidence[kind] = true
		await shot("native-" + kind + "-popup")
	assert(picker.get_popup().visible, "native popup did not open")
	var popup := picker.get_popup()
	var point: Vector2 = Vector2(popup.position) + Vector2(float(popup.size.x) * 0.5, float(popup.size.y) * (float(index) + 0.5) / float(popup.get_item_count()))
	operations.append({"native_popup":kind, "window_id":popup.get_window_id(), "position":str(popup.position), "size":str(popup.size), "row":index, "mouse_point":str(point), "method":"XTest OS mouse"})
	assert(OS.execute("python", ["/tmp/planet-os-input.py", "click", str(point.x), str(point.y)]) == 0)
	await create_timer(0.20).timeout
	await process_frame
	var current := resolve_picker(kind)
	assert(current.get_item_metadata(current.selected) == expected, "native choice differs")

func run() -> void:
	DirAccess.make_dir_recursive_absolute(OUT)
	assert(GDExtensionManager.load_extension("/tmp/planet-native-472/planet.gdextension") == GDExtensionManager.LOAD_STATUS_OK)
	game = load("res://src/main.gd").new()
	root.add_child(game)
	root.position = Vector2i.ZERO
	root.grab_focus()
	await process_frame
	print("STARTUP_ERROR ", game.startup_error)
	assert(game.startup_error.is_empty())
	game.set_process(false)
	game._skip_tutorial = true
	session = game.session
	var config: Dictionary = JSON.parse_string(FileAccess.get_file_as_string("res://generated/config.json"))
	var saved: Dictionary = session.campaign_system.save_state()
	for row in config.unlock.unlock: saved.grants[str(row.type)+":"+str(row.target)] = true
	record("synthetic_campaign_grants", session.campaign_system.restore_state(saved))
	record("crew_availability", {"ok": true, "changed":session.crew_system.refresh_availability(session.campaign_system.available)})
	var planets: Dictionary = session.planet_system.save_state()
	var a: Dictionary = planets.planets["1"]
	a.exploration = 240
	a.conquered = true
	for key in a.buildings:
		var row: Dictionary = {}
		for candidate in config.planet_build.planet_build:
			if str(candidate.id) == key: row = candidate
		a.buildings[key].state = "ready" if row.type == "refinery" else "built"
		a.buildings[key].work = int(row.build_explore)
	var b: Dictionary = planets.planets["2"]
	b.exploration = 22
	for key in b.buildings:
		var row: Dictionary = {}
		for candidate in config.planet_build.planet_build:
			if str(candidate.id) == key: row = candidate
		if row.type == "auto_explore":
			b.buildings[key].state = "built"
			b.buildings[key].work = int(row.build_explore)
		elif row.type == "refinery":
			b.buildings[key].state = "building"
			b.buildings[key].work = 2
	record("synthetic_planet_state",session.planet_system.restore_state(planets))
	session.planet_system.refresh_availability(session.campaign_system.available)
	var members: Array = session.crew_system.member_options()
	record("assign_explorer_1",session.coordinate_assignment(str(members[0].id),"planet_explore","1"))
	record("automatic_1",session.planet_system.set_automatic("1",true))
	record("assign_explorer_2",session.coordinate_assignment(str(members[1].id),"planet_explore","2"))
	var started: Dictionary = session.planet_system.save_state()
	started.planets["1"].elapsed = 24.0
	started.planets["2"].elapsed = 8.0
	record("synthetic_elapsed",session.planet_system.restore_state(started))
	session.planet_system.refresh_availability(session.campaign_system.available)
	var contributions: Dictionary = session.planet_system.contributions()
	record("publish_real_fixture_contributions",session.publish_world(contributions))
	game._sync_guidance()
	game._refresh_feature_visibility()
	await click(game._left_buttons.planet)
	panel = game.planet_scene
	panel.select_planet("1")
	panel.refresh()
	await create_timer(3).timeout
	await shot("01-default-planet1")
	await click(panel._facility_button)
	await shot("02-facilities-planet1-ready")
	await choose(panel.world_planet_picker,1)
	await create_timer(2).timeout
	await shot("03-same-drawer-planet2-building")
	assert(panel.selected_planet_id() == "2")
	assert(panel._snapshot.exploration == 22)
	for fact in panel._facts: assert(fact.planet_id == "2")
	assert(panel._drawer.visible and panel._facility_button.button_pressed)
	await choose(panel.world_planet_picker,0)
	await shot("04-switch-back-planet1-ready")
	await click(panel._claim_button)
	await process_frame
	await shot("05-planet1-claimed")

		if fact.id == panel._selected_facility: assert(fact.state == "built")

	await click(panel._cancel_button)
	await click(panel._crew_button)

	await choose(panel._assignment_picker,0)
	await click(panel._crew_button)
	await capture_motion()
	var file := FileAccess.open(OUT.path_join("operations.json"),FileAccess.WRITE)
	file.store_string(JSON.stringify(operations,"  "))
	file.close()
	if session.save_files != null: session.save_files.close(2000)
	game.queue_free()
	await process_frame
	await process_frame
	quit()

func mesh_rect(node: Node, camera: Camera3D, offset: Vector2) -> Rect2:
	var points: Array[Vector2] = []
	mesh_points(node, camera, offset, points)
	if points.is_empty(): return Rect2()
	var result := Rect2(points[0],Vector2.ZERO)
	for point in points: result = result.expand(point)
	return result

func mesh_points(node: Node, camera: Camera3D, offset: Vector2, points: Array[Vector2]) -> void:
	if node is MeshInstance3D and node.visible and node.mesh != null and not node.material_override is ShaderMaterial:
		var bounds: AABB = node.get_aabb()
		for index in 8:
			var v := bounds.position + bounds.size * Vector3(float(index & 1), float((index >> 1) & 1), float((index >> 2) & 1))
			points.append(camera.unproject_position(node.global_transform * v) + offset)
	for child in node.get_children(): mesh_points(child, camera, offset, points)

func rect_data(rect: Rect2) -> Array:
	return [rect.position.x, rect.position.y, rect.size.x, rect.size.y]

func sprite_rect(model: Node3D, sprite_name: String, camera: Camera3D, offset: Vector2) -> Rect2:
	var sprite := model.find_child(sprite_name, true, false) as Sprite3D
	assert(sprite != null, "Missing illustrated body: " + sprite_name)
	var center := camera.unproject_position(sprite.global_position) + offset
	var scale_factor := sprite.global_transform.basis.get_scale()
	var pixels_per_world := float(camera.get_viewport().size.y) / camera.size
	var extent := Vector2(sprite.texture.get_width() * scale_factor.x, sprite.texture.get_height() * scale_factor.y) * sprite.pixel_size * pixels_per_world
	return Rect2(center - extent * 0.5, extent)

func rect_gap(a: Rect2, b: Rect2) -> float:
	var dx := maxf(0.0, maxf(a.position.x - b.end.x, b.position.x - a.end.x))
	var dy := maxf(0.0, maxf(a.position.y - b.end.y, b.position.y - a.end.y))
	return Vector2(dx, dy).length()

func capture_motion() -> void:
	DirAccess.make_dir_recursive_absolute(OUT.path_join("motion-frames"))
	var timeline := FileAccess.open(OUT.path_join("motion.jsonl"), FileAccess.WRITE)
	var started := Time.get_ticks_usec()
	var previous_phase: float = panel._visual.globe.phases.x
	var previous_front := {}
	var turns := 0.0
	var next_key_turn := 0.10
	var frame_count := 0
	var key_count := 0
	var capture_usecs := 0
	var minimum_gap := INF
	var minimum_gap_frame := -1
	var last_seconds := 0.0
	while true:
		await RenderingServer.frame_post_draw
		var seconds := float(Time.get_ticks_usec() - started) / 1000000.0
		last_seconds = seconds
		var visual = panel._visual
		var architecture = visual.architecture
		var offset: Vector2 = architecture.get_global_position()
		var camera: Camera3D = architecture._camera
		var current: float = visual.globe.phases.x
		var delta := current - previous_phase
		if delta < -0.5: delta += 1.0
		turns += delta
		previous_phase = current
		if turns < 0.60:
			await create_timer(0.45).timeout
			continue
		var model_data := {}
		var rects := {}
		var horizon_crossing := false
		for kind in architecture._owned:
			var model: Node3D = architecture._models[kind]
			var bounds: Rect2
			match kind:
				"refinery": bounds = sprite_rect(model, "IllustratedRefineryBody", camera, offset)
				"equipment": bounds = sprite_rect(model, "IllustratedWorkshopBody", camera, offset)
				"shipyard": bounds = sprite_rect(model, "IllustratedShipyardBody", camera, offset)
				_: bounds = mesh_rect(model, camera, offset)
			rects[kind] = bounds
			var front: float = model.position.z
			if kind in ["equipment", "refinery"] and previous_front.has(kind):
				if front * float(previous_front[kind]) <= 0.0: horizon_crossing = true
			previous_front[kind] = front
			var parts := {}
			for part_name in architecture._parts[kind]:
				var part: Node3D = architecture._parts[kind][part_name]
				parts[part_name] = {"position": [part.position.x,part.position.y,part.position.z], "rotation": [part.rotation.x,part.rotation.y,part.rotation.z]}
			model_data[kind] = {"position": [model.position.x,model.position.y,model.position.z], "front_z": front, "screen_aabb": rect_data(bounds), "parts": parts}
		var gap := rect_gap(rects["auto_explore"], rects["shipyard"])
		if gap < minimum_gap:
			minimum_gap = gap
			minimum_gap_frame = frame_count
		var capture_started := Time.get_ticks_usec()
		var image := root.get_texture().get_image()
		var image_name := "motion-frames/frame_%04d.jpg" % frame_count
		image.save_jpg(OUT.path_join(image_name), 0.92)
		var key_due := frame_count == 0 or frame_count % 8 == 0 or horizon_crossing or turns >= 0.78
		if key_due:
			image.save_png(OUT.path_join("motion-frames/key_%04d.png" % frame_count))
			key_count += 1
			if turns >= next_key_turn: next_key_turn += 0.125
		capture_usecs += Time.get_ticks_usec() - capture_started
		var row := {"frame": frame_count, "file": image_name, "wall_seconds": seconds, "game_time": session.game_time, "visual_clock": visual.phase, "surface_phase": current, "surface_turns_since_start": turns, "capture_window": "left" if turns < 0.17 else "right", "cloud_phase": visual.globe.phases.y, "pause_reasons": session.pause_reasons, "models": model_data, "station_shipyard_conservative_rect_gap_px": gap, "horizon_crossing": horizon_crossing, "key_png": key_due}
		timeline.store_line(JSON.stringify(row))
		timeline.flush()
		assert(session.run_state.is_running())
		if turns >= 0.78 or seconds >= 180.0: break
		frame_count += 1
		await create_timer(0.45).timeout
	timeline.close()
	var summary := {"duration_wall_seconds": last_seconds, "frames": frame_count + 1, "surface_turns": turns, "full_cycle_complete": false, "key_png_frames": key_count, "capture_wall_seconds": float(capture_usecs) / 1000000.0, "station_shipyard_conservative_min_rect_gap_px": minimum_gap, "station_shipyard_min_gap_frame": minimum_gap_frame, "capture_method": "Godot 4.7.2 actual main framebuffer, local right 0.60–0.78 surface turns; 0.45s minimum sample interval; no engine frame cap"}
	var file := FileAccess.open(OUT.path_join("motion-summary.json"), FileAccess.WRITE)
	file.store_string(JSON.stringify(summary, "  "))
	file.close()
	print("MOTION_COMPLETE ", JSON.stringify(summary))
