extends SceneTree
func _initialize() -> void:
	call_deferred("run")
func frames(n: int) -> void:
	for i in n: await process_frame
func center(control: Control) -> Vector2:
	return control.get_global_transform_with_canvas() * (control.size * 0.5)
func pointer(point: Vector2, pressed: bool = false) -> void:
	var motion := InputEventMouseMotion.new()
	motion.position = point
	motion.global_position = point
	root.push_input(motion, false)
	if pressed:
		var button := InputEventMouseButton.new()
		button.button_index = MOUSE_BUTTON_LEFT
		button.position = point
		button.global_position = point
		button.pressed = true
		root.push_input(button, false)
func clear_guidance(session, game) -> void:
	var pending: Dictionary = session.pending_guidance()
	var tries := 0
	while not pending.is_empty() and tries < 20:
		assert(session.dismiss_guidance(str(pending.kind), str(pending.id)).ok)
		pending = session.pending_guidance()
		tries += 1
	game._sync_guidance()
func run() -> void:
	root.size = Vector2i(1440,900)
	root.content_scale_size = Vector2i(1440,900)
	var game: Node2D = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await frames(4)
	var session = game.session
	var saved: Dictionary = session.campaign_system.save_state()
	saved.grants["feature:reactor"] = true
	saved.grants["reactor_module:condensation"] = true
	assert(session.campaign_system.restore_state(saved).ok)
	session.reactor_system.refresh_availability(session.campaign_system.available)
	assert(session.reactor_system.refresh(session.reactor_inputs(), session.publish_reactor).ok)
	clear_guidance(session,game)
	game._select_left_module("reactor")
	await frames(5)
	var scene = game.reactor_scene
	clear_guidance(session,game)
	await frames(2)
	assert(scene._quote.ok and scene._quote.costs.has("2") and "2" in scene._quote.cost_resource_ids)
	assert(not scene._quote.can_purchase and scene._cost.text == "资金不足")
	assert(scene._cost_badge.visible and scene._cost_icon.texture != null)
	print("NO_FUNDS quote=",scene._quote," badge=",scene._cost_badge.visible," icon=",scene._cost_icon.texture != null," tint=",scene._cost_icon.modulate)
	root.get_texture().get_image().save_png("/tmp/reactor-material-no-funds.png")
	var amount: Dictionary = session.amount_api.apply("from_decimal",["1000"])
	assert(amount.ok)
	assert(session.economy.credit_yield("2",amount.value,session.wall_time,session.game_time).ok)
	await frames(5)
	clear_guidance(session,game)
	await frames(2)
	assert(scene._quote.ok and scene._quote.can_purchase and scene._quote.costs.has("2"))
	assert(scene._cost_badge.visible and scene._cost_icon.texture != null)
	print("FUNDED quote=",scene._quote," badge=",scene._cost_badge.visible," icon=",scene._cost_icon.texture != null," tint=",scene._cost_icon.modulate)
	root.get_texture().get_image().save_png("/tmp/reactor-material-funded.png")
	scene._purchase.grab_focus()
	await frames(2)
	assert(scene._purchase.has_focus())
	root.get_texture().get_image().save_png("/tmp/reactor-material-focus.png")
	scene._purchase.release_focus()
	scene._quantity.tooltip_text = ""
	var point := center(scene._quantity)
	pointer(point)
	await frames(2)
	assert(scene._quantity.is_hovered())
	root.get_texture().get_image().save_png("/tmp/reactor-material-hover.png")
	pointer(point, true)
	await frames(2)
	assert(scene._quantity.button_pressed)
	root.get_texture().get_image().save_png("/tmp/reactor-material-pressed.png")
	quit()
