extends SceneTree

const OUT := "/tmp/reactor-b01-interaction-"

func _initialize() -> void:
	call_deferred("run")

func frames(n: int) -> void:
	for i in n: await process_frame

func clear_guidance(session, game) -> void:
	var pending: Dictionary = session.pending_guidance()
	var tries := 0
	while not pending.is_empty() and tries < 20:
		assert(session.dismiss_guidance(str(pending.kind), str(pending.id)).ok)
		pending = session.pending_guidance()
		tries += 1
	game._sync_guidance()

func click(pos: Vector2) -> void:
	var press := InputEventMouseButton.new()
	press.button_index = MOUSE_BUTTON_LEFT
	press.pressed = true
	press.position = pos
	press.global_position = pos
	root.push_input(press, true)
	var release := InputEventMouseButton.new()
	release.button_index = MOUSE_BUTTON_LEFT
	release.pressed = false
	release.position = pos
	release.global_position = pos
	root.push_input(release, true)

func drag(start: Vector2, finish: Vector2) -> void:
	var press := InputEventMouseButton.new()
	press.button_index = MOUSE_BUTTON_LEFT
	press.pressed = true
	press.position = start
	press.global_position = start
	root.push_input(press, true)
	var motion := InputEventMouseMotion.new()
	motion.position = finish
	motion.global_position = finish
	motion.relative = finish - start
	motion.button_mask = MOUSE_BUTTON_MASK_LEFT
	root.push_input(motion, true)
	var release := InputEventMouseButton.new()
	release.button_index = MOUSE_BUTTON_LEFT
	release.pressed = false
	release.position = finish
	release.global_position = finish
	root.push_input(release, true)

func shot(name: String) -> void:
	await frames(3)
	await RenderingServer.frame_post_draw
	var result: Error = root.get_texture().get_image().save_png(OUT + name + "-1440x900.png")
	assert(result == OK)

func shot_fast(name: String) -> void:
	await RenderingServer.frame_post_draw
	var result: Error = root.get_texture().get_image().save_png(OUT + name + "-1440x900.png")
	assert(result == OK)

func read_row(model: Dictionary, id: String) -> Dictionary:
	for row in model.modules:
		if str(row.id) == id: return row
	return {}

func wallet_text(session) -> String:
	return str(session.amount_api.apply("to_display", [session.economy._wallet["2"]]).text)

func run() -> void:
	root.size = Vector2i(1440, 900)
	root.content_scale_size = Vector2i(1440, 900)
	var game: Node2D = load("res://src/main.tscn").instantiate()
	root.add_child(game)
	await frames(4)
	var session = game.session
	assert(session != null and game.startup_error.is_empty())
	assert(session.current_chrono_speed() == 1)
	var saved: Dictionary = session.campaign_system.save_state()
	saved.grants["feature:reactor"] = true
	saved.grants["reactor_module:condensation"] = true
	assert(session.campaign_system.restore_state(saved).ok)
	session.reactor_system.refresh_availability(session.campaign_system.available)
	assert(session.reactor_system.refresh(session.reactor_inputs(), session.publish_reactor).ok)
	clear_guidance(session, game)
	game._select_left_module("reactor")
	await frames(5)
	var scene = game.reactor_scene
	assert(scene != null and scene.visible)
	clear_guidance(session, game)
	await frames(3)
	var amount: Dictionary = session.amount_api.apply("from_decimal", ["1000"])
	assert(amount.ok)
	assert(session.economy.credit_yield("2", amount.value, session.wall_time, session.game_time).ok)
	await frames(3)
	var model: Dictionary = session.reactor_system.presentation_model()
	# Actual mouse click on Equal, followed by a grip drag to a valid smaller share.
	click(scene._equal.get_global_rect().get_center())
	await frames(6)
	model = session.reactor_system.presentation_model()
	assert(is_equal_approx(float(read_row(model, "weapons").ratio), 0.25))
	var slider: HSlider = scene._rows["weapons"].slider
	var rect: Rect2 = slider.get_global_rect()
	print("SLIDER rect=", rect, " value=", slider.value)
	var y: float = rect.position.y + rect.size.y * 0.5
	var start := Vector2(rect.position.x + 15.0 + 148.0 * 0.25, y)
	var valid := Vector2(rect.position.x + 15.0 + 148.0 * 0.10, y)
	drag(start, valid)
	await frames(8)
	model = session.reactor_system.presentation_model()
	var accepted: float = float(read_row(model, "weapons").ratio)
	assert(accepted > 0.04 and accepted < 0.23)
	assert(scene.last_result.get("ok", false))
	assert(absf(float(slider.value) / 10000.0 - accepted) < 0.011)
	print("MOUSE_ACCEPT ratio=", accepted, " slider=", slider.value, " allocated=", model.allocated, " notice=", scene._notice.text)
	await shot("mouse-accepted")
	var commands_before: int = int(scene.presentation_counts.commands)
	var denied_start := Vector2(rect.position.x + 15.0 + 148.0 * accepted, y)
	var denied_end := Vector2(rect.position.x + 15.0 + 148.0 * 0.85, y)
	drag(denied_start, denied_end)
	await frames(8)
	model = session.reactor_system.presentation_model()
	print("REJECT_STATE before=", accepted, " actual=", read_row(model, "weapons").ratio, " slider=", slider.value, " last=", scene.last_result, " commands=", int(scene.presentation_counts.commands)-commands_before); assert(is_equal_approx(float(read_row(model, "weapons").ratio), accepted))
	assert(absf(float(slider.value) / 10000.0 - accepted) < 0.011)
	assert(str(scene.last_result.get("reason", "")) == "ReactorPlanOverCapacity")
	assert(int(scene.presentation_counts.commands) <= commands_before + 1)
	print("MOUSE_REJECT reason=", scene.last_result.get("reason"), " actual=", accepted, " slider=", slider.value, " commands_delta=", int(scene.presentation_counts.commands) - commands_before, " notice=", scene._notice.text)
	await shot("mouse-rejected")
	# The quantity and upgrade controls are also activated with GUI mouse events.
	click(scene._quantity.get_global_rect().get_center())
	await frames(4)
	assert(scene._quantity.text == "×10" and int(scene._quote.count) == 10 and scene._quote.can_purchase)
	var ten_quote: Dictionary = scene._quote.duplicate(true)
	var before_level: int = int(session.reactor_system.presentation_model().level)
	var before_wallet: Dictionary = session.economy._wallet["2"].duplicate(true)
	click(scene._purchase.get_global_rect().get_center())
	await frames(2)
	var after_level: int = int(session.reactor_system.presentation_model().level)
	assert(after_level == before_level + 10 and scene.last_result.get("ok",false))
	assert(scene._notice.text == "已升级 10 级")
	var ten_cost: Dictionary = session.amount_api.apply("from_decimal", [str(ten_quote.costs["2"])])
	assert(ten_cost.ok)
	var ten_expected: Dictionary = session.amount_api.apply("sub_nonnegative", [before_wallet, ten_cost.value])
	assert(ten_expected.ok and session.amount_api.apply("compare", [ten_expected.value, session.economy._wallet["2"]]).scalar == 0)
	print("TEN_PURCHASE count=10 cost=", ten_quote.costs["2"], " level=", before_level, "->", after_level, " wallet_before=", session.amount_api.apply("to_display", [before_wallet]).text, " wallet_after=", wallet_text(session), " feedback=", scene._notice.text)
	await shot_fast("ten-purchased")
	# At level 11, one accepted unit is a visible positive share below 1%.
	var low_plan := {"weapons":"1", "defence":"0", "smelting":"0", "condensation":"0"}
	assert(session.reactor_system.apply_allocation_plan(low_plan, session.reactor_inputs(), session.publish_reactor).ok)
	await frames(6)
	model = session.reactor_system.presentation_model()
	assert(float(read_row(model, "weapons").load) > 0.0 and float(model.load) > 0.0)
	assert(scene._rows["weapons"].ratio.text == "<1%")
	print("LOW load=", model.load, " weapon=", read_row(model, "weapons").load, " ratio_label=", scene._rows["weapons"].ratio.text, " stage=", scene._stage._paint.get_shader_parameter("core_load"))
	await shot("low")
	low_plan["weapons"] = "0"
	assert(session.reactor_system.apply_allocation_plan(low_plan, session.reactor_inputs(), session.publish_reactor).ok)
	await frames(6)
	model = session.reactor_system.presentation_model()
	assert(float(read_row(model, "weapons").load) == 0.0 and float(model.load) == 0.0)
	assert(scene._rows["weapons"].ratio.text == "0%")
	print("ZERO load=", model.load, " ratio_label=", scene._rows["weapons"].ratio.text, " stage=", scene._stage._paint.get_shader_parameter("core_load"))
	await shot("true-zero")
	# Restore budget through the public economy API to exercise MAX after the ×10 transaction.
	assert(session.economy.credit_yield("2", amount.value, session.wall_time, session.game_time).ok)
	await frames(3)
	click(scene._quantity.get_global_rect().get_center())
	await frames(5)
	assert(scene._quantity.text == "MAX" and scene._quote.ok and scene._quote.can_purchase)
	var max_quote: Dictionary = scene._quote.duplicate(true)
	before_level = int(session.reactor_system.presentation_model().level)
	before_wallet = session.economy._wallet["2"].duplicate(true)
	click(scene._purchase.get_global_rect().get_center())
	await frames(2)
	after_level = int(session.reactor_system.presentation_model().level)
	assert(after_level == before_level + int(max_quote.count) and scene.last_result.get("ok",false))
	assert(scene._notice.text == "已升级 " + str(max_quote.count) + " 级")
	var max_cost: Dictionary = session.amount_api.apply("from_decimal", [str(max_quote.costs["2"])])
	assert(max_cost.ok)
	var max_expected: Dictionary = session.amount_api.apply("sub_nonnegative", [before_wallet, max_cost.value])
	assert(max_expected.ok and session.amount_api.apply("compare", [max_expected.value, session.economy._wallet["2"]]).scalar == 0)
	print("MAX_PURCHASE count=", max_quote.count, " cost=", max_quote.costs["2"], " level=", before_level, "->", after_level, " wallet_before=", session.amount_api.apply("to_display", [before_wallet]).text, " wallet_after=", wallet_text(session), " feedback=", scene._notice.text)
	await shot_fast("max-purchased")
	quit()
