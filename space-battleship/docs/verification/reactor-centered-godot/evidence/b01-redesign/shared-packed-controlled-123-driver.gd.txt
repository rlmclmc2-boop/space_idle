extends SceneTree

const OUT := "/tmp/reactor-packed-shared-interaction-"

func _initialize() -> void:
	call_deferred("run")

func frames(n: int) -> void:
	for i in n: await process_frame

func clear_guidance(session, game) -> void:
	var pending: Dictionary = session.pending_guidance()
	var tries := 0
	while not pending.is_empty() and tries < 20:
		assert(session.dismiss_guidance(str(pending.kind), str(pending.id)).ok)
		pending = session.pending_guidance()
		tries += 1
	game._sync_guidance()

func click(pos: Vector2) -> void:
	var press := InputEventMouseButton.new()
	press.button_index = MOUSE_BUTTON_LEFT
	press.pressed = true
	press.position = pos
	press.global_position = pos
	root.push_input(press, true)
	var release := InputEventMouseButton.new()
	release.button_index = MOUSE_BUTTON_LEFT
	release.pressed = false
	release.position = pos
	release.global_position = pos
	root.push_input(release, true)

func drag(start: Vector2, finish: Vector2) -> void:
	var press := InputEventMouseButton.new()
	press.button_index = MOUSE_BUTTON_LEFT
	press.pressed = true
	press.position = start
	press.global_position = start
	root.push_input(press, true)
	var motion := InputEventMouseMotion.new()
	motion.position = finish
	motion.global_position = finish
	motion.relative = finish - start
	motion.button_mask = MOUSE_BUTTON_MASK_LEFT
	root.push_input(motion, true)
	var release := InputEventMouseButton.new()
	release.button_index = MOUSE_BUTTON_LEFT
	release.pressed = false
	release.position = finish
	release.global_position = finish
	root.push_input(release, true)

func shot(name: String) -> void:
	await frames(3)
	await RenderingServer.frame_post_draw
	var result: Error = root.get_texture().get_image().save_png(OUT + name + "-1440x900.png")
	assert(result == OK)

func shot_fast(name: String) -> void:
	await RenderingServer.frame_post_draw
	var result: Error = root.get_texture().get_image().save_png(OUT + name + "-1440x900.png")
	assert(result == OK)

func read_row(model: Dictionary, id: String) -> Dictionary:
	for row in model.modules:
		if str(row.id) == id: return row
	return {}

func wallet_text(session) -> String:
	return str(session.amount_api.apply("to_display", [session.economy._wallet["2"]]).text)

func wait_seconds(seconds: float, session, game, scene, label: String) -> void:
	var before: float = float(scene._stage.visual_phase())
	var game_before: float = session.game_time
	await create_timer(seconds).timeout
	clear_guidance(session,game)
	var delta: float = float(scene._stage.visual_phase())-before
	print("SEGMENT_END ",label," wall_requested=",seconds," game_delta=",session.game_time-game_before," phase_delta=",delta," pending=",session.pending_guidance())
	assert((is_zero_approx(delta) if label=="true_zero" else delta > seconds*0.65) and session.pending_guidance().is_empty())

func run() -> void:
	root.size=Vector2i(1440,900);root.content_scale_size=Vector2i(1440,900)
	var game: Node2D=load("res://src/main.tscn").instantiate();root.add_child(game);await frames(80)
	var session=game.session
	assert(session!=null and game.startup_error.is_empty() and session.current_chrono_speed()==1)
	var saved: Dictionary=session.campaign_system.save_state()
	saved.grants["feature:reactor"]=true;saved.grants["reactor_module:condensation"]=true
	assert(session.campaign_system.restore_state(saved).ok)
	session.reactor_system.refresh_availability(session.campaign_system.available)
	assert(session.reactor_system.refresh(session.reactor_inputs(),session.publish_reactor).ok)
	clear_guidance(session,game);game._refresh_feature_visibility();game._zone_shell._route_buttons["reactor"].pressed.emit();await frames(5)
	var scene: Control=game.reactor_scene;assert(scene.visible)
	# Wait for the real first pickup prompt, then dismiss it through Session before capture.
	var preroll_start:=Time.get_ticks_msec()
	while not session.tutorial.completed("pickup") and Time.get_ticks_msec()-preroll_start<30000:
		await process_frame
		clear_guidance(session,game)
	assert(session.tutorial.completed("pickup"))
	clear_guidance(session,game);await frames(4);clear_guidance(session,game)
	assert(session.pending_guidance().is_empty() and not game._guidance_panel.visible)
	print("CAPTURE_CONTEXT pid=",OS.get_process_id()," host=051cbd0d module=6d0ee146 size=",root.size," speed=",session.current_chrono_speed()," pickup_seen=",session.tutorial.save_state().seen.get("pickup",false))
	FileAccess.open("/tmp/reactor-state-sequence-ready",FileAccess.WRITE).store_string("ready\n")
	while not FileAccess.file_exists("/tmp/reactor-state-sequence-start"):
		await process_frame
		clear_guidance(session,game)
	var Reactor=preload("res://src/systems/reactor.gd")
	for count in [1,2,3]:
		var module_ids: String="smelting" if count==1 else "weapons,smelting" if count==2 else "weapons,defence,smelting"
		var parameters: Dictionary=session.reactor_system._parameters.duplicate(true)
		parameters.reactorModules=module_ids
		var fixture_owner=Reactor.new(parameters,session.amount_api,session.run_state)
		var fixture_publish: Callable=func(_factors,_wallet,_kind):
			fixture_owner.accept()
			return {"ok":true}
		fixture_owner.refresh_availability(func(kind,target):return kind=="feature" and target=="reactor")
		assert(fixture_owner.refresh(session.reactor_inputs(),fixture_publish).ok)
		scene._owner=fixture_owner;scene._publish=fixture_publish
		scene.bind_presentation(fixture_owner.presentation_model,func():return {"game_time":session.game_time,"paused":false,"running":true})
		scene.bind_purchase_quote(func(quantity,maximum):return fixture_owner.purchase_max_preview(session.economy.can_pay) if maximum else fixture_owner.purchase_preview(session.economy.can_pay,quantity))
		fixture_owner.presentation_changed.connect(scene.invalidate_presentation)
		assert(fixture_owner.distribute(session.reactor_inputs(),fixture_publish).ok)
		await frames(5);clear_guidance(session,game)
		var model: Dictionary=fixture_owner.presentation_model()
		assert(model.ok and model.modules.size()==count and is_equal_approx(float(model.load),1.0))
		print("SEGMENT_START controlled=",count," ids=",fixture_owner.enabled_modules()," loads=",model.modules.map(func(row):return [row.id,row.load])," phase=",scene._stage.visual_phase()," game_time=",session.game_time)
		await wait_seconds(3.0,session,game,scene,"controlled_"+str(count))
	# Restore the real Session owner and native scene callbacks for the purchase chain.
	scene._owner=session.reactor_system;scene._publish=session.publish_reactor
	scene.bind_presentation(func():return session.reactor_system.presentation_model(),func():return {"game_time":session.game_time,"paused":not session.pause_reasons.is_empty() or not session.run_state.is_running(),"running":session.run_state.is_running()})
	scene.bind_purchase_quote(func(quantity,maximum):return session.reactor_system.purchase_max_preview(session.economy.can_pay) if maximum else session.reactor_system.purchase_preview(session.economy.can_pay,quantity))
	scene.invalidate();await frames(5);clear_guidance(session,game)
	var amount: Dictionary=session.amount_api.apply("from_decimal",["1000"])
	assert(amount.ok and session.economy.credit_yield("2",amount.value,session.wall_time,session.game_time).ok)
	await frames(4)
	assert(session.reactor_system.distribute(session.reactor_inputs(),session.publish_reactor).ok);await frames(6)
	var model4: Dictionary=session.reactor_system.presentation_model()
	print("FOUR_RESTORE ",JSON.stringify(model4)," scene_ids=",scene._ids," last=",scene.last_result);assert(model4.ok and model4.modules.size()==4 and is_equal_approx(float(model4.load),1.0))
	print("SEGMENT_START four_upgrade_before level=",model4.level," wallet=",wallet_text(session)," loads=",model4.modules.map(func(row):return [row.id,row.load])," phase=",scene._stage.visual_phase())
	await wait_seconds(1.6,session,game,scene,"four_upgrade_before")
	click(scene._quantity.get_global_rect().get_center());await frames(3)
	assert(scene._quantity.text=="×10" and int(scene._quote.count)==10 and scene._quote.can_purchase)
	var quote: Dictionary=scene._quote.duplicate(true)
	var before_wallet: Dictionary=session.economy._wallet["2"].duplicate(true)
	var before_level: int=int(session.reactor_system.presentation_model().level)
	click(scene._purchase.get_global_rect().get_center());await frames(2)
	var after_level: int=int(session.reactor_system.presentation_model().level)
	var cost: Dictionary=session.amount_api.apply("from_decimal",[str(quote.costs["2"])])
	var expected: Dictionary=session.amount_api.apply("sub_nonnegative",[before_wallet,cost.value])
	assert(cost.ok and expected.ok and after_level==before_level+10 and scene.last_result.get("ok",false))
	assert(session.amount_api.apply("compare",[expected.value,session.economy._wallet["2"]]).scalar==0)
	print("SEGMENT_START purchase_result level=",before_level,"->",after_level," cost=",quote.costs["2"]," wallet=",wallet_text(session)," notice=",scene._notice.text," phase=",scene._stage.visual_phase())

	var low_model: Dictionary=session.reactor_system.presentation_model()
