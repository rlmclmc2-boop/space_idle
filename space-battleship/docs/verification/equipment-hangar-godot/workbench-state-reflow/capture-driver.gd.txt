extends SceneTree
var app
var panel
var host
var records: Array = []
var failures: Array = []
var destination: String = "/workspace/space_idle/space-battleship/docs/verification/equipment-hangar-godot/workbench-state-reflow"
func _initialize() -> void:
    call_deferred("run")
func check(value: bool, label: String) -> void:
    if not value: failures.append(label); printerr("FAIL: ", label)
func flush() -> void:
    for i in 4: await process_frame
    await RenderingServer.frame_post_draw
func screenshot(name: String) -> void:
    await flush()
    root.get_texture().get_image().save_png(destination.path_join(name + ".png"))
func click(control: Control) -> void:
    var point: Vector2 = control.get_global_rect().get_center()
    var down := InputEventMouseButton.new()
    down.position = point; down.global_position = point; down.button_index = MOUSE_BUTTON_LEFT; down.pressed = true
    Input.parse_input_event(down)
    await process_frame
    var up := InputEventMouseButton.new()
    up.position = point; up.global_position = point; up.button_index = MOUSE_BUTTON_LEFT; up.pressed = false
    Input.parse_input_event(up)
    await flush()
func snapshot() -> Dictionary:
    return {"wallet": host.economy.budget().duplicate(true), "wallet_text": host.economy.balance_rows(), "slots": host.equipment_slot_facts(), "native": host.battle.hud_values(), "game_time": host.game_time, "pause_reasons": host.pause_reasons.duplicate(), "candidate": panel._candidate, "selected_slot": panel._selected, "comparison": panel.last_comparison.duplicate(true), "quantity": panel._quantity, "maximum": panel._maximum}
func wallet(iron: String, material: String) -> void:
    var state: Dictionary = host.economy.save_state()
    state.wallet = {"1": host.amount_api.apply("from_decimal", [iron]).value, "2": host.amount_api.apply("from_decimal", ["1000"]).value, "degenerate_matter": host.amount_api.apply("from_decimal", [material]).value}
    check(host.economy.restore_state(state).ok, "validated controlled wallet setup")
    host.economy.publish_budget_change()
func capture_state(name:String)->void:
    await flush()
    var frame:Rect2=panel._workbench.get_global_rect()
    for control in [panel._purchase,panel._cost,panel._current_value,panel._next_value,panel._candidate_row,panel._feedback]:check(frame.encloses(control.get_global_rect()),name+" visible control inside shared workbench")
    var candidates:Dictionary={}
    for id in panel._choices:
        var item:Dictionary=panel._choices[id]
        candidates[id]={"visible":item.button.is_visible_in_tree(),"rect":item.button.get_global_rect(),"disabled":item.button.disabled,"mouse_filter":item.button.mouse_filter,"focus_mode":item.button.focus_mode}
    records.append({"action":name,"frame":frame,"layout":panel._workbench_layout.duplicate(true),"candidates":candidates,"primary_surface":panel._upgrade_data.get_global_rect(),"transaction_rail":panel._transaction_rail.get_global_rect(),"purchase":panel._purchase.get_global_rect(),"comparison":panel._comparison_group.get_global_rect(),"state":snapshot()})
    await screenshot(name)

    var locked_before:Dictionary=snapshot()
    await click(panel._choices.cannon.button)

    await capture_state("01-fresh-same-single-blank-locked")

    for i in range(1,100):c.state.cleared_stages[i]=true

    wallet("1234567890123456789012345678901234567890","9999999999999999999999");panel._dirty=true;await flush()

    await click(panel._choices.laser.button);await capture_state("04-long-max-same-multi")
    var counts:Dictionary=panel.presentation_counts.duplicate();host.economy.display_changed.emit();await flush()
    for key in ["refreshes","comparison_reads","slot_fact_reads","equipment_option_reads","scene_redraw_requests"]:check(panel.presentation_counts[key]==counts[key],"same-budget signal avoids unrelated work "+key)
    await click(panel._quantity_buttons[0]);var before_purchase:Dictionary=snapshot();await click(panel._purchase)
    var receipt:Dictionary=panel.last_result.get("purchase_receipt",{})
    for resource in receipt.get("spent",{}):
        var expected:Dictionary=host.amount_api.apply("sub_nonnegative",[before_purchase.wallet[resource],receipt.spent[resource]])
        check(host.amount_api.apply("compare",[expected.value,host.economy.budget()[resource]]).scalar==0,"actual moved purchase exact wallet minus receipt "+str(resource))
