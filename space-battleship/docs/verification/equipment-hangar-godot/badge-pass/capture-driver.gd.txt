    await RenderingServer.frame_post_draw

    var point:Vector2=control.get_global_rect().get_center()
    for pressed in [true,false]:
        var event:=InputEventMouseButton.new();event.position=point;event.global_position=point;event.button_index=MOUSE_BUTTON_LEFT;event.pressed=pressed;Input.parse_input_event(event);await process_frame
    await flush()

    check(app.startup_error.is_empty() and host.run_state.is_running() and not host.pause_reasons.has("recovery"),"actual fresh startup")
    if not failures.is_empty():app._close_game();return
    host.set_pause("user",true)
    var campaign:Dictionary=host.campaign_system.save_state()

    check(host.campaign_system.change_ship("Heavy_Battleship",host.commit_ship_refit).ok,"actual Heavy refit")
    for pair in [["weapon:1","missile"],["weapon:2","longLaser"],["weapon:3","cannon"],["weapon:4","laser"],["defence:0","shield"],["defence:1","armour"],["defence:2","shield"]]:
        check(host.equipment.equip_slot(pair[0],pair[1],host.equipment_inputs(),host.publish_equipment_refit).ok,"real setup equip "+pair[0])
    await flush()

        await click(panel._slot_buttons[fact.id].button)
        check(panel._selected==fact.id and panel.last_comparison.current.equipment_id==fact.equipment_id,"real GUI selection "+fact.id)
        var target:Rect2=panel._component_rect(fact).grow(5)
        var badge:Rect2=panel._locator_badge_rect(fact,target)

        var terminals:Dictionary={}

        rows.append({"slot_id":fact.id,"badge_text":text,"badge_rect":badge,"component_rect":target,"terminals":terminals,"comparison":panel.last_comparison.duplicate(true)})
        root.get_texture().get_image().save_png(destination.path_join("selected-"+str(fact.id).replace(":","-")+".png"))
