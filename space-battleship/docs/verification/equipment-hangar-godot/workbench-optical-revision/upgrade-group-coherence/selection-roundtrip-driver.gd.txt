extends SceneTree
var app
var panel
var host
var records: Array = []
var failures: Array = []
var destination: String = "/tmp/equipment-upgrade-group-final"
func _initialize() -> void:
    call_deferred("run")
func check(value: bool, label: String) -> void:
    if not value: failures.append(label); printerr("FAIL: ", label)
func flush() -> void:
    for i in 4: await process_frame
    await RenderingServer.frame_post_draw
func screenshot(name: String) -> void:
    await flush()
    var pixels:Image=root.get_texture().get_image()
    pixels.save_png(destination.path_join(name + ".png"))
    var region:Rect2i=Rect2i(panel._workbench.get_global_rect().grow(4)).intersection(Rect2i(0,0,pixels.get_width(),pixels.get_height()))
    if name=="08-batch":region=Rect2i(panel._batch_frame.get_global_rect().grow(4))
    if name in ["01-weapon-ready","03-weapon-disabled","05-armour-replacement-ready","08-batch"]:
        pixels.get_region(region).save_png(destination.path_join(name+"-local.png"))
    records.append({"name":name,"state":snapshot(),"purchase_disabled":panel._purchase.disabled,"install_visible":panel._install.visible,"install_disabled":panel._install.disabled,"current_title":panel._current_title.text,"purchase_text":panel._purchase.text,"install_text":panel._install.text,"install_tooltip":panel._install.tooltip_text,"cooldown_text":panel._switch_state.text,"comparison_headers":panel._compare_headers.map(func(n):return n.text),"candidate_status_texts":panel._choices.values().map(func(item):return item.status.text),"fee_caption":panel._cost.text,"upgrade_group":upgrade_group(),"candidate_rect":panel._candidate_row.get_global_rect(),"purchase_rect":panel._purchase.get_global_rect(),"candidate_nodes":panel._choices.keys(),"presentation_counts":panel.presentation_counts.duplicate(true)})
func click(control: Control) -> void:
    var point: Vector2 = control.get_global_rect().get_center()
    var down := InputEventMouseButton.new()
    down.position = point; down.global_position = point; down.button_index = MOUSE_BUTTON_LEFT; down.pressed = true
    Input.parse_input_event(down)
    await process_frame
    var up := InputEventMouseButton.new()
    up.position = point; up.global_position = point; up.button_index = MOUSE_BUTTON_LEFT; up.pressed = false
    Input.parse_input_event(up)
    await flush()
func snapshot() -> Dictionary:
    return {"wallet": host.economy.budget().duplicate(true), "wallet_text": host.economy.balance_rows(), "slots": host.equipment_slot_facts(), "native": host.battle.hud_values(), "game_time": host.game_time, "pause_reasons": host.pause_reasons.duplicate(), "candidate": panel._candidate, "selected_slot": panel._selected, "comparison": panel.last_comparison.duplicate(true), "quantity": panel._quantity, "maximum": panel._maximum}
func wallet(iron: String, material: String) -> void:
    var state: Dictionary = host.economy.save_state()
    state.wallet = {"1": host.amount_api.apply("from_decimal", [iron]).value, "2": host.amount_api.apply("from_decimal", ["1000"]).value, "degenerate_matter": host.amount_api.apply("from_decimal", [material]).value}
    check(host.economy.restore_state(state).ok, "validated controlled wallet setup")
    host.economy.publish_budget_change()
func local_state(name:String, batch:bool=false)->void:
    await flush()
    var rect:Rect2=panel._batch_frame.get_global_rect() if batch else panel._workbench.get_global_rect()
    root.get_texture().get_image().get_region(Rect2i(rect.grow(4))).save_png(destination.path_join(name+".png"))
func upgrade_group()->Dictionary:
    return {"quantity_node":panel._quantity_button.get_instance_id(),"purchase_node":panel._purchase.get_instance_id(),"quantity_style":panel._quantity_button.get_theme_stylebox("normal").get_instance_id(),"purchase_style":panel._purchase.get_theme_stylebox("normal").get_instance_id(),"quantity_rect":panel._quantity_button.get_global_rect(),"purchase_rect":panel._purchase.get_global_rect(),"quantity_size":panel._quantity_button.size,"purchase_size":panel._purchase.size,"gap":panel._purchase.global_position.x-panel._quantity_button.get_global_rect().end.x,"purchase_visible":panel._purchase.is_visible_in_tree(),"quantity_visible":panel._quantity_button.is_visible_in_tree(),"comparison_visible":panel._comparison_group.is_visible_in_tree(),"main_current":panel._current_value.text,"main_next":panel._next_value.text,"main_next_caption":panel._upgrade_caption.text,"fee_visible":panel._cost.is_visible_in_tree(),"fee_cells_visible":panel._cost_cells.values().filter(func(c):return c.amount.is_visible_in_tree()).size(),"workbench_rect":panel._workbench.get_global_rect(),"candidate_rect":panel._candidate_row.get_global_rect(),"candidate_card_sizes":panel._choices.values().map(func(c):return c.button.size)}
func run()->void:
    app=load("res://src/main.tscn").instantiate();root.add_child(app);host=app.session;panel=app.equipment_scene
    host.set_pause("user",true);await flush();await create_timer(1.2).timeout;await flush()
    var c:Dictionary=host.campaign_system.save_state();c.state.highest_reached=100;c.cycle_reached=100;c.state.state="menu";c.state.cleared_stages={}
    for i in range(1,100):c.state.cleared_stages[i]=true
    host.campaign_system.restore_state(c);host.equipment.refresh_availability(host.campaign_system.available,100)
    var e:Dictionary=host.equipment.save_state()
    for row in e.slots:
        if row.id=="weapon:0":row.paid_level=19;row.switch_ready_at=0.0
    host.equipment.restore_state(e,host.campaign_system.ship_facts().mounts);host.publish_equipment_refit(host.economy.budget(),"refit","",null,false)
    wallet("100000","1000");panel._dirty=true;await flush();await create_timer(1.2).timeout;await flush()
    var results:Array=[]
    await click(panel._quantity_button);await flush()
    check(panel._quantity==10 and not panel._maximum,"quote mode10 prepared without purchase")
    for pair in [["weapon:0","missile","laser"],["defence:0","shield","armour"]]:
        await click(panel._slot_buttons[pair[0]].button);await flush()
        var before:Dictionary={"group":upgrade_group(),"quantity":panel._quantity,"maximum":panel._maximum,"wallet":host.economy.budget().duplicate(true)}
        await click(panel._choices[pair[1]].button);await flush()
        var selected:Dictionary={"group":upgrade_group(),"quantity":panel._quantity,"maximum":panel._maximum}
        await click(panel._choices[pair[2]].button);await flush()
        var returned:Dictionary={"group":upgrade_group(),"quantity":panel._quantity,"maximum":panel._maximum,"wallet":host.economy.budget().duplicate(true)}
        check(returned.quantity==before.quantity and returned.maximum==before.maximum and returned.wallet==before.wallet,"mode and wallet preserved for "+pair[0])
        for key in ["quantity_node","purchase_node","quantity_style","purchase_style","quantity_size","purchase_size","gap"]:check(before.group[key]==selected.group[key] and before.group[key]==returned.group[key],"round trip same group "+pair[0]+key)
        results.append({"slot":pair[0],"before":before,"selected":selected,"returned":returned})
    FileAccess.open(destination.path_join("selection-roundtrip.json"),FileAccess.WRITE).store_string(JSON.stringify({"source_commit":"304883fbde448cd32fa288ae97967e43100209bc","records":results,"failures":failures,"scope":"Actual GUI current-candidate-current selection with quote10 retained; no screenshots, purchase, installation or independent transaction matrix"},"  "))
    print("EQUIPMENT_UPGRADE_SELECTION_ROUNDTRIP_COMPLETE failures=",failures.size())
    panel=null;host=null;app.queue_free();app=null
    for i in 8:await process_frame
    quit()
