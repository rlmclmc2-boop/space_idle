extends SceneTree
var app
var panel
var host
var records: Array = []
var failures: Array = []
var destination: String = "/tmp/equipment-painted-blend"
func _initialize() -> void:
    call_deferred("run")
func check(value: bool, label: String) -> void:
    if not value: failures.append(label); printerr("FAIL: ", label)
func flush() -> void:
    for i in 4: await process_frame
    await RenderingServer.frame_post_draw
func screenshot(name: String) -> void:
    await flush()
    root.get_texture().get_image().save_png(destination.path_join(name + ".png"))
func click(control: Control) -> void:
    var point: Vector2 = control.get_global_rect().get_center()
    var down := InputEventMouseButton.new()
    down.position = point; down.global_position = point; down.button_index = MOUSE_BUTTON_LEFT; down.pressed = true
    Input.parse_input_event(down)
    await process_frame
    var up := InputEventMouseButton.new()
    up.position = point; up.global_position = point; up.button_index = MOUSE_BUTTON_LEFT; up.pressed = false
    Input.parse_input_event(up)
    await flush()
func snapshot() -> Dictionary:
    return {"wallet": host.economy.budget().duplicate(true), "wallet_text": host.economy.balance_rows(), "slots": host.equipment_slot_facts(), "native": host.battle.hud_values(), "game_time": host.game_time, "pause_reasons": host.pause_reasons.duplicate(), "candidate": panel._candidate, "selected_slot": panel._selected, "comparison": panel.last_comparison.duplicate(true), "quantity": panel._quantity, "maximum": panel._maximum}
func wallet(iron: String, material: String) -> void:
    var state: Dictionary = host.economy.save_state()
    state.wallet = {"1": host.amount_api.apply("from_decimal", [iron]).value, "2": host.amount_api.apply("from_decimal", ["1000"]).value, "degenerate_matter": host.amount_api.apply("from_decimal", [material]).value}
    check(host.economy.restore_state(state).ok, "validated controlled wallet setup")
    host.economy.publish_budget_change()
func run()->void:
    app=load("res://src/main.tscn").instantiate();root.add_child(app);host=app.session;panel=app.equipment_scene
    host.set_pause("user",true);await flush()
    var c:Dictionary=host.campaign_system.save_state();c.state.highest_reached=100;c.cycle_reached=100;c.state.state="menu";c.state.cleared_stages={}
    for i in range(1,100):c.state.cleared_stages[i]=true
    host.campaign_system.restore_state(c);host.equipment.refresh_availability(host.campaign_system.available,100)
    var e:Dictionary=host.equipment.save_state()
    for row in e.slots:
        if row.id=="weapon:0":row.paid_level=19;row.switch_ready_at=0.0
    host.equipment.restore_state(e,host.campaign_system.ship_facts().mounts);host.publish_equipment_refit(host.economy.budget(),"refit","",null,false)
    wallet("100000","1000");panel._dirty=true;await flush();await screenshot("02-double")
    await click(panel._choices.missile.button)
    panel._open_batch();await flush();await screenshot("04-batch")
    print("MATERIAL_PREVIEW_COMPLETE");app.queue_free();await process_frame;quit()
