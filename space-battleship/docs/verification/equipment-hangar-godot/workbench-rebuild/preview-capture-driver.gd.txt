func flush() -> void:

func click(control: Control) -> void:
    var point: Vector2 = control.get_global_rect().get_center()
    var down := InputEventMouseButton.new()
    down.position = point; down.global_position = point; down.button_index = MOUSE_BUTTON_LEFT; down.pressed = true
    Input.parse_input_event(down)
    await process_frame
    var up := InputEventMouseButton.new()
    up.position = point; up.global_position = point; up.button_index = MOUSE_BUTTON_LEFT; up.pressed = false
    Input.parse_input_event(up)
    await flush()

func monochrome_ramp(colors:Array) -> bool:
    # Antialiased/253-alpha source pixels blend the one silhouette RGB with its disabled backing.
    var foreground:=Vector3(12,37,48)
    var background:=Vector3(49,67,74)
    var direction:Vector3=background-foreground
    for html in colors:
        var color:=Color.from_string("#"+str(html),Color.MAGENTA)
        var rgb:=Vector3(color.r,color.g,color.b)*255.0
        var t:float=clampf((rgb-foreground).dot(direction)/direction.length_squared(),0,1)
        var distance:Vector3=(rgb-(foreground+direction*t)).abs()
        if maxf(distance.x,maxf(distance.y,distance.z))>2:return false
    return true
func candidate_pixels(id: String) -> Dictionary:
    var item:Dictionary=panel._choices[id]
    var atlas:AtlasTexture=item.icon.texture
    var picture:Image=atlas.atlas.get_image().get_region(Rect2i(atlas.region))
    var rect:Rect2=item.icon.get_global_rect()
    var factor:float=minf(rect.size.x/picture.get_width(),rect.size.y/picture.get_height())
    var origin:Vector2=rect.get_center()-Vector2(picture.get_size())*factor*0.5
    var viewport_image:Image=root.get_texture().get_image()
    var colors:Dictionary={};var sampled:int=0
    for y in range(ceili(origin.y),floori(origin.y+picture.get_height()*factor)):
        for x in range(ceili(origin.x),floori(origin.x+picture.get_width()*factor)):
            var source:Vector2i=Vector2i((Vector2(x+0.5,y+0.5)-origin)/factor)
            var opaque:bool=true
            for offset in [Vector2i.ZERO,Vector2i(-4,-4),Vector2i(4,4),Vector2i(-4,4),Vector2i(4,-4)]:
                var point:Vector2i=source+offset
                if point.x<0 or point.y<0 or point.x>=picture.get_width() or point.y>=picture.get_height() or picture.get_pixelv(point).a<0.98:opaque=false;break
            if opaque:
                sampled+=1;colors[viewport_image.get_pixel(x,y).to_html(false)]=true
    return {"id":id,"disabled":item.button.disabled,"material_is_silhouette":item.icon.material==panel._locked_icon_material,"sampled_opaque_pixels":sampled,"opaque_rgb_colors":colors.keys(),"state_text":item.status.text}
func locked_state(label: String) -> void:
    await flush()
    var items:Array=[]
    for id in ["laser","missile","cannon","longLaser"]:
        var row:Dictionary=candidate_pixels(id)
        check(row.sampled_opaque_pixels>20,label+" actual candidate pixels sampled "+id)
        if row.disabled:
            check(row.material_is_silhouette and monochrome_ramp(row.opaque_rgb_colors),label+" locked pixels contain no internal RGB detail "+id)
        else:
            check(not row.material_is_silhouette and row.opaque_rgb_colors.size()>10,label+" unlocked official color detail retained "+id)
        items.append(row)
    records.append({"action":label,"candidates":items,"state":snapshot()})
func ship_record(label: String) -> void:
    var bounds:Rect2=Rect2(panel._hull_visible_bounds[panel._ship])
    var draw:Rect2=panel._ship_rect()
    var factor:float=draw.size.x/768.0
    var visible:=Rect2(draw.position+bounds.position*factor,bounds.size*factor)
    var envelope:Rect2=visible
    for row in host.equipment_slot_facts():
        if row.dormant or row.category!="weapon" or str(row.equipment_id).is_empty():continue
        envelope=envelope.merge(panel._component_rect(row))
        if row.category=="weapon":
            var point:Array=panel._catalog[panel._ship].mounts[int(row.index)]
            var expected:Vector2=draw.position+Vector2(point[0],point[1])*factor
            check(expected.distance_to(panel._mount_position(row))<0.001,label+" official source mount follows assembly "+str(row.id))
    var selected:Dictionary=panel._selected_fact()
    if not selected.is_empty():
        var badge:Rect2=panel._locator_badge_rect(selected,panel._component_rect(selected).grow(5))
        if badge.has_area():envelope=envelope.merge(badge)
    var safe:=Rect2(224,158,560,446)
    check(safe.encloses(envelope),label+" whole current assembly has hangar clearance")
    var aft:Vector2=draw.position+panel._preview_reference[panel._ship].aft_source*factor
    check(aft.distance_to(panel._preview_reference[panel._ship].dock_target)<0.01,label+" visible aft anchor rests at authored pad docking line")
    records.append({"action":label,"ship":panel._ship,"visible_hull":visible,"assembly_envelope":envelope,"safe":safe,"source_reference":panel._preview_reference[panel._ship],"canvas_rect":draw,"same_assembly_zoom":panel._preview_zooms[panel._ship],"state":snapshot()})
func tree(node:Node,rows:Array) -> void:
    if node is Control:
        var row:Dictionary={"path":str(node.get_path()),"class":node.get_class(),"visible":node.is_visible_in_tree(),"rect":node.get_global_rect(),"clip":node.clip_contents}
        if node is Label:row.text=node.text
        if node is TextureRect:
            row.texture=node.texture.resource_path if node.texture!=null else ""
            if node.texture is AtlasTexture:row.atlas_source=node.texture.atlas.resource_path;row.region=node.texture.region
        rows.append(row)
    for child in node.get_children():tree(child,rows)
