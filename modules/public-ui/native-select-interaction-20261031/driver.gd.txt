extends SceneTree
func _initialize() -> void: call_deferred("run")
func frames(n:int) -> void:
 for i in n: await process_frame
func shot(name:String) -> void:
 await RenderingServer.frame_post_draw
 root.get_texture().get_image().save_png("/tmp/space-idle-runtime472/select-interaction/"+name+".png")
func clear_guidance(session,game) -> void:
 var pending:Dictionary=session.pending_guidance()
 var tries:=0
 while not pending.is_empty() and tries<20:
  assert(session.dismiss_guidance(str(pending.kind),str(pending.id)).ok)
  pending=session.pending_guidance()
  tries+=1
 game._sync_guidance()
func click(point:Vector2) -> void:
 Input.warp_mouse(point)
 await frames(2)
 for pressed in [true,false]:
  var event:=InputEventMouseButton.new()
  event.button_index=MOUSE_BUTTON_LEFT
  event.pressed=pressed
  event.position=point
  event.global_position=point
  root.push_input(event)
  await frames(2)
func key(popup:PopupMenu,code:Key) -> void:
 for pressed in [true,false]:
  var event:=InputEventKey.new()
  event.keycode=code
  event.pressed=pressed
  Input.parse_input_event(event)
  await frames(2)
func run() -> void:
 root.size=Vector2i(1440,900)
 root.content_scale_size=Vector2i(1440,900)
 var game:Node2D=load("res://src/main.tscn").instantiate()
 root.add_child(game)
 await frames(45)
 assert(game.session!=null and game.startup_error.is_empty())
 clear_guidance(game.session,game)
 game._refresh_feature_visibility()
 game._zone_shell._route_buttons["settings"].pressed.emit()
 await frames(8)
 clear_guidance(game.session,game)
 var select:OptionButton=game._damage_option
 var popup:=select.get_popup()
 print("BEFORE selected=",select.selected," text=",select.text," option_rect=",select.get_global_rect())
 await shot("before-click")
 await click(select.get_global_rect().get_center())
 print("OPEN visible=",popup.visible," pos=",popup.position," size=",popup.size," focused=",popup.get_focused_item())
 await shot("clicked-open")
 assert(popup.visible)
 popup.grab_focus()
 await frames(2)
 print("POPUP_HAS_FOCUS ",popup.has_focus())
 await key(popup,KEY_DOWN)
 await key(popup,KEY_DOWN)
 print("DOWN focused=",popup.get_focused_item()," selected=",select.selected)
 await shot("keyboard-down")
 await key(popup,KEY_ENTER)
 print("ENTER visible=",popup.visible," selected=",select.selected," text=",select.text," mode=",game._damage_mode)
 await shot("keyboard-confirmed")
 if game.session.save_files!=null: game.session.save_files.close(2000)
 game.queue_free()
 await frames(3)
 quit()
